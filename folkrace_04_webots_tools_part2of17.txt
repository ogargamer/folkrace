class BasePlannerGenome:
    """Generic genome wrapper providing serialization, clamping, and repair."""

    def __init__(self, specs: List[Tuple[str, float, float, type, float, str]], values: Optional[Dict[str, Any]] = None):
        self.specs = specs
        self.values: Dict[str, Any] = {}
        for name, min_v, max_v, vtype, def_v, _ in specs:
            if values and name in values:
                self.values[name] = vtype(values[name])
            elif values and name == "w_gap_clear" and "alt_lookahead_depth_weight" in values:
                self.values[name] = vtype(values["alt_lookahead_depth_weight"])
            else:
                self.values[name] = vtype(def_v)
        self.clamp_and_repair()

    def to_dict(self) -> Dict[str, Any]:
        return dict(self.values)

    def to_array(self) -> np.ndarray:
        arr = []
        for name, _, _, _, _, _ in self.specs:
            arr.append(float(self.values[name]))
        return np.array(arr, dtype=np.float64)

    def from_array(self, arr: np.ndarray):
        for idx, (name, min_v, max_v, vtype, _, _) in enumerate(self.specs):
            if idx < len(arr):
                val = arr[idx]
                if vtype is int:
                    val = int(round(val))
                self.values[name] = vtype(val)
        self.clamp_and_repair()

    def clamp_and_repair(self):
        for name, min_v, max_v, vtype, _, _ in self.specs:
            val = self.values[name]
            val = max(min_v, min(max_v, val))
            if vtype is int:
                val = int(round(val))
            elif vtype is float:
                if min_v == max_v:
                    # Preserve full precision for fixed physical constants (e.g. dt=0.016667, recovery_steer=0.5623)
                    val = float(min_v)
                else:
                    val = round(float(val), 3)
            self.values[name] = vtype(val)

    def get_bounds(self) -> Tuple[np.ndarray, np.ndarray]:
        lower = np.array([s[1] for s in self.specs], dtype=np.float64)
        upper = np.array([s[2] for s in self.specs], dtype=np.float64)
        return lower, upper

def load_best_planner_genome(
    filepath: Optional[str] = None,
    genome_cls: Optional[Any] = None,
) -> Tuple[Optional[BasePlannerGenome], float, Dict[str, Any]]:
    """
    Loads chromosome and metrics from best_planner_genome.json (or specified filepath).
    Returns (genome_instance, fitness, metrics_dict).
    """
    path = filepath or BEST_GENOME_PATH
    if not os.path.exists(path):
        return None, -1e9, {}
    try:
        with open(path, "r", encoding="utf-8") as f:
            data = json.load(f)
        genes = data.get("genes") or data.get("best_genome") or data
        fitness = float(data.get("fitness", data.get("best_fitness", -1e9)))
        metrics = data.get("metrics", data.get("best_metrics", {}))
        cls = genome_cls or get_genome_class()
        genome = cls(genes)
        return genome, fitness, metrics
    except Exception as e:
        print(f"[Genome Loader] Warning: Failed to load genome from {path}: {e}", flush=True)
        return None, -1e9, {}

SPEED_PHYSICAL_MAX_MPS: float = 3.84
DISTANCE_VALID_MIN_M: float = 0.10


def compute_scalar_fitness(
    metrics: Dict[str, Any],
    gene_dict: Optional[Dict[str, Any]] = None,
    eval_duration_s: Optional[float] = None,
) -> float:
    """Computes duration-normalized scalar optimization fitness from Webots trial metrics.

    Normalizes distance and lap metrics relative to FITNESS_BASE_DURATION_S (25.0s) so that
    longer simulation runs (e.g. 50s-60s) do not inflate the fitness score. When a candidate
    survives longer runs without crashing or stalling, a modest bonus (up to +10% at
    EVAL_DURATION_MAX_S) is applied to reward extended collision-free survival.
    """
    dist_m = float(metrics.get("distance_m", 0.0))
    mean_speed = float(metrics.get("mean_speed_mps", 0.0))
    laps = int(metrics.get("laps_completed", 0))
    crashed = bool(metrics.get("crashed", False))
    stuck = bool(metrics.get("stalled", False))

    if math.isnan(dist_m) or dist_m <= DISTANCE_VALID_MIN_M:
        return -5000.0

    if math.isnan(mean_speed) or mean_speed < 0.0:
        mean_speed = 0.0
    elif mean_speed > SPEED_PHYSICAL_MAX_MPS:
        mean_speed = SPEED_PHYSICAL_MAX_MPS

    # Determine planned trial evaluation duration for normalization
    trial_dur = eval_duration_s
    if trial_dur is None:
        trial_dur = metrics.get("eval_duration_s") or metrics.get("target_duration_s")
    if trial_dur is None:
        # Fallback to duration_s; if crashed/stuck early, guard against small sim_time inflating distance
        raw_dur = float(metrics.get("duration_s", FITNESS_BASE_DURATION_S))
        trial_dur = max(FITNESS_BASE_DURATION_S, raw_dur) if (crashed or stuck) else raw_dur
    trial_dur = max(1.0, float(trial_dur))

    # Duration normalization: scale distance and laps to baseline 25.0s equivalent
    norm_scale = FITNESS_BASE_DURATION_S / trial_dur
    effective_dist_m = dist_m * norm_scale
    effective_laps = laps * norm_scale

    base_score = (
        (effective_dist_m * FITNESS_PROGRESS_WEIGHT)
        + (mean_speed * FITNESS_SPEED_WEIGHT)
        + (effective_laps * FITNESS_LAP_BONUS)
    )

    if crashed:
        return float(base_score - FITNESS_COLLISION_PENALTY)
    elif stuck:
        return float(base_score - FITNESS_STALL_PENALTY)
    else:
        # Survival bonus: up to +10% for completing longer evaluations without crashing
        dur_excess = max(0.0, trial_dur - FITNESS_BASE_DURATION_S)
        dur_span = max(1.0, EVAL_DURATION_MAX_S - FITNESS_BASE_DURATION_S)
        bonus_factor = min(1.0, dur_excess / dur_span)
        duration_bonus_mult = 1.0 + (FITNESS_DURATION_BONUS_MAX * bonus_factor)
        return float(base_score * duration_bonus_mult)

  3. Evaluates all 84 candidates across 14 headless Webots workers in parallel.
  4. Records metrics, reports any candidates beating 10.2k and the champion, and

def build_tight_population_84() -> list:
    """Builds 84 individuals clustered tightly around the >= 10.2k parameters."""
    # 1. Load 10.874k champion
    champ_path = os.path.join(CURR_DIR, "best_planner_genome.json")
    with open(champ_path, "r", encoding="utf-8") as f:
        champ_json = json.load(f)
    champ_genes = champ_json["genes"]

    # 2. Load 10.259k CMA-ES best
    cma_path = os.path.join(CURR_DIR, "cma_state.pkl")
    with open(cma_path, "rb") as f:
        cma_data = pickle.load(f)
    es = cma_data["es"]

    g_dummy = Option2SLMPCGenome()
    lower, upper = g_dummy.get_bounds()
    span = upper - lower

    u_best = es.result.xbest
    x_cma = lower + u_best * span
    g_cma = Option2SLMPCGenome()
    g_cma.from_array(x_cma)
    cma_genes = g_cma.to_dict()

    pop = []

    # Candidate 0: Exact 10.874k Champion
    pop.append(Option2SLMPCGenome(champ_genes))
    # Candidate 1: Exact 10.259k CMA Best
    pop.append(Option2SLMPCGenome(cma_genes))

    arr_champ = Option2SLMPCGenome(champ_genes).to_array()
    arr_cma = Option2SLMPCGenome(cma_genes).to_array()

    # Candidates 2-15: Direct linear interpolations between 10.8k and 10.2k
    for alpha in np.linspace(0.05, 0.95, 14):
        interp = (1.0 - alpha) * arr_champ + alpha * arr_cma
        g = Option2SLMPCGenome()
        g.from_array(interp)
        pop.append(g)

    # Candidates 16-49: Ultra-tight Gaussian mutations around 10.8k Champion (sigma 0.5% - 2.5% span)
    np.random.seed(42)
    for sig in [0.005, 0.010, 0.015, 0.020, 0.025]:
        for _ in range(7):
            noise = np.random.normal(0.0, sig, size=span.shape) * span
            cand = np.clip(arr_champ + noise, lower, upper)
            g = Option2SLMPCGenome()
            g.from_array(cand)
            pop.append(g)

    # Candidates 50-83: Ultra-tight Gaussian mutations around 10.2k CMA Best (sigma 0.5% - 2.5% span)
    for sig in [0.005, 0.010, 0.015, 0.020, 0.025]:
        for _ in range(7):
            noise = np.random.normal(0.0, sig, size=span.shape) * span
            cand = np.clip(arr_cma + noise, lower, upper)
            g = Option2SLMPCGenome()
            g.from_array(cand)
            pop.append(g)

    # Truncate or pad to exactly 84
    while len(pop) < 84:
        noise = np.random.normal(0.0, 0.01, size=span.shape) * span
        cand = np.clip(arr_champ + noise, lower, upper)
        g = Option2SLMPCGenome()
        g.from_array(cand)
        pop.append(g)

    return pop[:84]

def build_gen1_population(target_pop_size: int = 42) -> list:
    """Builds Gen 1 population with the 32 historical candidate genomes + tight mutations."""
    champ_genome, champ_fit, champ_metrics = load_best_planner_genome()
    lower, upper = champ_genome.get_bounds()
    span = upper - lower
    span[span <= 1e-9] = 1.0

    # Load checkpoint population
    ckpt_path = os.path.join(CURR_DIR, "ga_checkpoint.json")
    pop_dicts = []
    if os.path.exists(ckpt_path):
        with open(ckpt_path, "r", encoding="utf-8") as f:
            ckpt = json.load(f)
            pop_dicts = ckpt.get("population", [])

    pop = []
    # Candidate 0: Champion
    pop.append(champ_genome)

    # Candidates 1..31: Distinct individuals from the checkpoint population
    for ind_dict in pop_dicts:
        if len(pop) >= 32:
            break
        # Avoid duplicate champion
        g = Option2SLMPCGenome(ind_dict)
        if len(pop) == 1 or np.max(np.abs(g.to_array() - champ_genome.to_array())) > 1e-4:
            pop.append(g)

    # If we need more up to 32, create tight micro-interpolations around champion
    arr_champ = champ_genome.to_array()
    np.random.seed(42)
    while len(pop) < 32:
        noise = np.random.normal(0.0, 0.02, size=span.shape) * span
        cand_arr = np.clip(arr_champ + noise, lower, upper)
        g = Option2SLMPCGenome()
        g.from_array(cand_arr)
        pop.append(g)

    # Pad remaining 10 candidates with tight sigma=0.05 mutations to reach pop_size=42
    while len(pop) < target_pop_size:
        noise = np.random.normal(0.0, 0.05, size=span.shape) * span
        cand_arr = np.clip(arr_champ + noise, lower, upper)
        g = Option2SLMPCGenome()
        g.from_array(cand_arr)
        pop.append(g)

    return pop[:target_pop_size]

def test_fitness_computation():
    print("Testing Fitness Function & Duration Scaling...")
    clean_metrics = {"distance_m": 22.5, "mean_speed_mps": 2.2, "laps_completed": 1, "crashed": False, "stalled": False}
    fit_clean = compute_scalar_fitness(clean_metrics)
    assert fit_clean > 2000.0
    print(f"  -> Clean run fitness (22.5m, 1 lap, default 25s): {fit_clean:.1f}")

    crash_metrics = {"distance_m": 5.0, "mean_speed_mps": 1.5, "laps_completed": 0, "crashed": True, "stalled": False}
    fit_crash = compute_scalar_fitness(crash_metrics)
    print(f"  -> Crashed run fitness (5.0m): {fit_crash:.1f}")
    assert fit_crash < fit_clean and fit_crash > 0.0

    # 1. Verify duration normalization: longer sim without crashing doesn't inflate ~2.4x, but gets ~10% bonus
    # Baseline 25s run at 2.5 m/s
    m_25s = {"distance_m": 62.5, "mean_speed_mps": 2.5, "laps_completed": 4, "crashed": False, "stalled": False, "eval_duration_s": 25.0}
    f_25s = compute_scalar_fitness(m_25s)

    # 60s run at identical speed (2.5 m/s) -> 2.4x distance and laps (150m, 10 laps)
    m_60s = {"distance_m": 150.0, "mean_speed_mps": 2.5, "laps_completed": 10, "crashed": False, "stalled": False, "eval_duration_s": 60.0}
    f_60s = compute_scalar_fitness(m_60s)

    # Without normalization f_60s would be ~24,000 (+140% inflation).
    # With normalization + 10% bonus: f_60s should be ~1.10x of f_25s!
    ratio = f_60s / f_25s
    print(f"  -> Duration Scaling: 25s clean = {f_25s:.1f}, 60s clean = {f_60s:.1f} (Ratio: {ratio:.3f}x, expected ~1.100x)")
    assert 1.09 <= ratio <= 1.11, f"Expected ~1.10x ratio, got {ratio:.4f}"

    # 50s run at identical speed -> 2.0x distance and laps (125m, 8 laps)
    m_50s = {"distance_m": 125.0, "mean_speed_mps": 2.5, "laps_completed": 8, "crashed": False, "stalled": False, "eval_duration_s": 50.0}
    f_50s = compute_scalar_fitness(m_50s)
    ratio_50 = f_50s / f_25s
    print(f"  -> Duration Scaling: 50s clean = {f_50s:.1f} (Ratio: {ratio_50:.3f}x, expected ~1.071x)")
    assert 1.06 <= ratio_50 <= 1.08, f"Expected ~1.071x ratio, got {ratio_50:.4f}"

    # 2. Verify crashed runs: no survival bonus awarded, normalized distance stays consistent
    m_crash_25s = {"distance_m": 31.25, "mean_speed_mps": 2.5, "laps_completed": 2, "crashed": True, "stalled": False, "eval_duration_s": 25.0}
    m_crash_60s = {"distance_m": 75.0, "mean_speed_mps": 2.5, "laps_completed": 5, "crashed": True, "stalled": False, "eval_duration_s": 60.0}
    f_crash_25s = compute_scalar_fitness(m_crash_25s)
    f_crash_60s = compute_scalar_fitness(m_crash_60s)
    diff_crash = abs(f_crash_60s - f_crash_25s)
    print(f"  -> Halfway crash: 25s trial = {f_crash_25s:.1f}, 60s trial = {f_crash_60s:.1f} (Difference: {diff_crash:.2f})")
    assert diff_crash <= 20.0, "Normalized crashed fitness across different durations should closely match within lap rounding!"

# ROS 2 & Google Cartographer SLAM (Experimental / Simulation Mapping)

> [!NOTE]
> **Hardware Context**: This module serves as an experimental simulation reference for 2D SLAM and offline track map generation. The physical Folkrace racing vehicle operates with **zero LiDAR**, relying exclusively on a solid-state quad VL53L8CX Time-of-Flight array and Curvilinear SL-MPC.

This package provides an integration between Webots simulation, ROS 2 Humble, and Google Cartographer 2D SLAM.

## 4. Webots to ROS 2 Cartographer Architecture

The Webots robot has:
- **LiDAR**: `ms200k_lidar` (360 FOV, 450 points, 10-60 Hz)
- **IMU**: `bno08x_imu` (Orientation/Quaternion) + `accelerometer` + `gyro`
- **Frame Tree**: `map` -> `odom` -> `base_link` -> `laser` & `imu_link`

### Communication Pipeline:
1. **Webots Python Controller Bridge**: Connects to Webots controller API, reads `ms200k_lidar`, `gyro`, `accelerometer`, and `bno08x_imu`, and publishes ROS 2 standard topics:
   - `/scan` (`sensor_msgs/msg/LaserScan`)
   - `/imu` (`sensor_msgs/msg/Imu`)
   - `/odom` (`nav_msgs/msg/Odometry` or via TF odometry)
   - `/tf` (`tf2_msgs/msg/TFMessage`) for `base_link` -> `laser` & `imu_link`
2. **Google Cartographer Node**: Subscribes to `/scan` and `/imu`, performs 2D submap scan matching and loop closure, and outputs `/map` and optimized poses.

## 6. Webots ROS 2 Publisher Node (`webots_webots/robotics/ros2_cartographer_bridge.py`)

A clean bridge running in ROS 2 / Webots Python controller:

```python
import rclpy
from rclpy.node import Node
from sensor_msgs.msg import LaserScan, Imu
from geometry_msgs.msg import TransformStamped
import tf2_ros
import math
import time

class WebotsCartographerBridge(Node):
    def __init__(self):
        super().__init__('webots_cartographer_bridge')
        
        self.scan_pub = self.create_publisher(LaserScan, '/scan', 10)
        self.imu_pub = self.create_publisher(Imu, '/imu', 50)
        self.tf_broadcaster = tf2_ros.StaticTransformBroadcaster(self)
        
        self.publish_static_transforms()
        self.get_logger().info("Webots ROS 2 Cartographer Bridge Initialized.")

    def publish_static_transforms(self):
        # Base Link -> Laser Transform
        t_laser = TransformStamped()
        t_laser.header.stamp = self.get_clock().now().to_msg()
        t_laser.header.frame_id = "base_link"
        t_laser.child_frame_id = "laser"
        t_laser.transform.translation.x = 0.0
        t_laser.transform.translation.y = 0.01
        t_laser.transform.translation.z = 0.0725
        t_laser.transform.rotation.w = 1.0

        # Base Link -> IMU Link Transform
        t_imu = TransformStamped()
        t_imu.header.stamp = self.get_clock().now().to_msg()
        t_imu.header.frame_id = "base_link"
        t_imu.child_frame_id = "imu_link"
        t_imu.transform.translation.x = 0.0
        t_imu.transform.translation.y = -0.01
        t_imu.transform.translation.z = 0.015
        t_imu.transform.rotation.w = 1.0

        self.tf_broadcaster.sendTransform([t_laser, t_imu])

    def publish_lidar_scan(self, range_image, fov_rad, min_range, max_range, stamp_sec):
        scan_msg = LaserScan()
        scan_msg.header.stamp.sec = int(stamp_sec)
        scan_msg.header.stamp.nanosec = int((stamp_sec - int(stamp_sec)) * 1e9)
        scan_msg.header.frame_id = "laser"
        
        num_readings = len(range_image)
        scan_msg.angle_min = -fov_rad / 2.0
        scan_msg.angle_max = fov_rad / 2.0
        scan_msg.angle_increment = fov_rad / num_readings
        scan_msg.time_increment = 0.0
        scan_msg.scan_time = 0.02
        scan_msg.range_min = float(min_range)
        scan_msg.range_max = float(max_range)
        scan_msg.ranges = [float(r) for r in range_image]
        
        self.scan_pub.publish(scan_msg)

    def publish_imu(self, accel_xyz, gyro_xyz, quat_xyzw, stamp_sec):
        imu_msg = Imu()
        imu_msg.header.stamp.sec = int(stamp_sec)
        imu_msg.header.stamp.nanosec = int((stamp_sec - int(stamp_sec)) * 1e9)
        imu_msg.header.frame_id = "imu_link"
        
        imu_msg.orientation.x = float(quat_xyzw[0])
        imu_msg.orientation.y = float(quat_xyzw[1])
        imu_msg.orientation.z = float(quat_xyzw[2])
        imu_msg.orientation.w = float(quat_xyzw[3])
        
        imu_msg.angular_velocity.x = float(gyro_xyz[0])
        imu_msg.angular_velocity.y = float(gyro_xyz[1])
        imu_msg.angular_velocity.z = float(gyro_xyz[2])
        
        imu_msg.linear_acceleration.x = float(accel_xyz[0])
        imu_msg.linear_acceleration.y = float(accel_xyz[1])
        imu_msg.linear_acceleration.z = float(accel_xyz[2])
        
        self.imu_pub.publish(imu_msg)
```

#!/usr/bin/env python3
"""
================================================================================
  MAP CONTENT VALIDATOR & NON-EMPTY CHECKER
  Verifies whether an exported PNG map contains rendered track data (non-empty).
================================================================================
"""

import sys
import os
import numpy as np
from PIL import Image

def check_map_file(png_path):
    if not os.path.exists(png_path):
        return False, "File does not exist"
    
    if os.path.getsize(png_path) < 1000:
        return False, "File size too small (<1KB)"

    try:
        im = Image.open(png_path).convert('RGB')
        arr = np.array(im)
        # Check standard deviation across channels/pixels to detect blank/solid frames
        std_dev = np.std(arr)
        unique_colors = len(np.unique(arr.reshape(-1, arr.shape[2]), axis=0))
        
        if std_dev < 1.0 or unique_colors < 10:
            return False, f"Blank image (std_dev={std_dev:.2f}, unique_colors={unique_colors})"
        
        return True, f"Valid map image (unique_colors={unique_colors}, std_dev={std_dev:.2f})"
    except Exception as e:
        return False, f"Image read error: {e}"

if __name__ == '__main__':
    if len(sys.argv) < 2:
        print("Usage: python check_map_empty.py <path_to_png>")
        sys.exit(1)
    
    is_valid, msg = check_map_file(sys.argv[1])
    print(f"[{'VALID' if is_valid else 'EMPTY'}] {sys.argv[1]}: {msg}")
    sys.exit(0 if is_valid else 1)
