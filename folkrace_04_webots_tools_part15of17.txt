def _diff_dicts(a: Dict[str, Any], b: Dict[str, Any], prefix: str = "") -> List[Dict[str, Any]]:
    diffs: List[Dict[str, Any]] = []
    for k in sorted(set(a) | set(b)):
        key_path = f"{prefix}.{k}" if prefix else str(k)
        va, vb = a.get(k), b.get(k)
        if isinstance(va, dict) and isinstance(vb, dict):
            diffs.extend(_diff_dicts(va, vb, key_path))
        elif va != vb:
            diffs.append({"key": key_path, "a": va, "b": vb})
    return diffs

def format_window_table(rows: List[Dict[str, Any]]) -> str:
    if not rows:
        return "(no frames in range)"
    header = "  fid |  sim_t | mode                    | v_cmd / real | str c /  a | f_tof |  slip | plan_ms | shield        | gap     | flags"
    sep = "-" * len(header)
    out = [header, sep]
    for r in rows:
        fl = ",".join(r["reasons"]) if r["reasons"] else "-"
        out.append(
            f"{r['frame_id']:5d} | {r['sim_time']:6.2f} | {r['mode']:<23} | "
            f"{r['v_cmd']:5.2f} /{r['v_real']:5.2f} | {r['steer_cmd_deg']:5.1f} /{r['steer_actual_deg']:5.1f} | "
            f"{r['f_tof_min']:5.2f} | {r['slip_ratio']:5.2f} | {r['plan_ms']:7.2f} | "
            f"{r['shield']:<13} | {r['gap']:<7} | {fl}"
        )
    return "\n".join(out)

#!/usr/bin/env python3
"""
================================================================================
FOLKRACE BROWSER LATENCY & TELEMETRY STREAM PROFILER
================================================================================
Measures client-side latency, arrival jitter, packet delivery consistency,
and rendering pipeline timing exactly as experienced by a browser (e.g. Firefox).

Features:
  1. WebSocket Streaming Benchmark (ws://robot:8080/ws_live):
     - Sub-millisecond RFC 6455 packet arrival intervals.
     - Frame arrival jitter and deviation from nominal 20 Hz (50.0 ms) target.
     - Stutter detection (intervals > 66.7 ms) and hitches (> 100.0 ms).
     - End-to-end client latency (Flight-to-Glass: Robot TS -> Client Arrival -> Parse).
     - Full client pipeline simulation: Header unpack + Zlib decompression + Canvas LUT.
     - High-precision percentiles: Min, P50, P90, P95, P99, P99.9, and Max.
  2. HTTP Polling Benchmark (http://robot:8080/api/live_bin & /api/live):
     - Measures Request-Response RTT distribution across polling frequencies.
     - Quantifies the jitter and TCP connection pooling penalties of HTTP fallback.
  3. Real In-Browser Firefox Harness (--firefox):
     - Automatically launches Firefox, connects to the Tactical Mission Control GUI,
       and collects live performance.now() metrics directly from the browser DOM.

Usage:
  python tools/measure_browser_latency.py                      # Full side-by-side benchmark (10s)
  python tools/measure_browser_latency.py --mode ws            # WebSocket streaming benchmark only
  python tools/measure_browser_latency.py --mode http          # HTTP polling benchmark only
  python tools/measure_browser_latency.py --firefox            # Run benchmark inside real Firefox
  python tools/measure_browser_latency.py --duration 15        # Benchmark over 15 seconds
  python tools/measure_browser_latency.py --host 192.168.20.120 # Specify target Pi host
================================================================================
"""

def resolve_host(preferred: Optional[str] = None, port: int = DEFAULT_PORT) -> Tuple[str, str]:
    """Auto-detects active Raspberry Pi debug server host and base URL."""
    candidates = [preferred] if preferred else DEFAULT_CANDIDATES
    for h in candidates:
        if not h:
            continue
        try:
            url = f"http://{h}:{port}/api/status"
            req = urllib.request.Request(url, headers={"User-Agent": "FolkraceLatencyTester/1.0"})
            with urllib.request.urlopen(req, timeout=1.5) as resp:
                if resp.status == 200:
                    return h, f"http://{h}:{port}"
        except Exception:
            continue
    # Fallback to preferred or first candidate
    fallback = preferred or DEFAULT_CANDIDATES[0]
    return fallback, f"http://{fallback}:{port}"


def compute_percentiles(arr: List[float]) -> Dict[str, float]:
    """Computes standard latency percentiles for an array of float measurements."""
    if not arr:
        return {"count": 0, "min": 0.0, "mean": 0.0, "p50": 0.0, "p90": 0.0, "p95": 0.0, "p99": 0.0, "p999": 0.0, "max": 0.0}
    s = sorted(arr)
    n = len(s)
    return {
        "count": n,
        "min": s[0],
        "mean": statistics.mean(s),
        "p50": s[int(n * 0.50)],
        "p90": s[min(n - 1, int(n * 0.90))],
        "p95": s[min(n - 1, int(n * 0.95))],
        "p99": s[min(n - 1, int(n * 0.99))],
        "p999": s[min(n - 1, int(n * 0.999))],
        "max": s[-1],
    }


def benchmark_websocket_stream(
    host: str,
    port: int = DEFAULT_PORT,
    duration_s: float = 10.0,
    target_hz: float = 20.0,
    verbose: bool = True,
) -> Dict[str, Any]:
    """
    Connects to the Pi 4 WebSocket endpoint (ws://host:port/ws_live) and records
    inter-frame arrival intervals, jitter, packet sizes, client parsing, and
    simulated canvas rendering durations.
    """
    if verbose:
        print(f"\n[WebSocket Profiler] Connecting to ws://{host}:{port}/ws_live...")

    sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    sock.setsockopt(socket.IPPROTO_TCP, socket.TCP_NODELAY, 1)
    sock.settimeout(3.0)

    t_connect0 = time.perf_counter()
    try:
        sock.connect((host, port))
    except Exception as e:
        if verbose:
            print(f"[WebSocket Error] Failed to connect to {host}:{port}: {e}")
        return {"error": str(e), "connected": False}

    # RFC 6455 Client Handshake
    sec_key = base64.b64encode(os.urandom(16)).decode()
    handshake = (
        f"GET /ws_live HTTP/1.1\r\n"
        f"Host: {host}:{port}\r\n"
        f"Upgrade: websocket\r\n"
        f"Connection: Upgrade\r\n"
        f"Sec-WebSocket-Key: {sec_key}\r\n"
        f"Sec-WebSocket-Version: 13\r\n\r\n"
    )
    sock.sendall(handshake.encode("utf-8"))

    # Read HTTP Response
    buf = b""
    while b"\r\n\r\n" not in buf:
        chunk = sock.recv(1024)
        if not chunk:
            sock.close()
            return {"error": "Server closed during handshake", "connected": False}
        buf += chunk

    hdr_data, buf = buf.split(b"\r\n\r\n", 1)
    dt_handshake = (time.perf_counter() - t_connect0) * 1000.0
    status_line = hdr_data.decode("utf-8", errors="ignore").splitlines()[0]
    if "101" not in status_line:
        sock.close()
        return {"error": f"Bad handshake response: {status_line}", "connected": False}

    if verbose:
        print(f"[WebSocket Profiler] Connected in {dt_handshake:.1f} ms. Streaming for {duration_s:.1f}s...")

    # Pre-generate LUT for simulated browser canvas construction
    try:
        import numpy as np
        has_numpy = True
        lut = np.zeros((512, 4), dtype=np.uint8)
        lut[:97] = [14, 22, 34, 150]
        lut[97:166] = [245, 215, 30, 130]
        lut[166:256] = [255, 0, 0, 255]
        lut[256:] = [255, 255, 0, 200]
    except ImportError:
        has_numpy = False

    intervals_ms: List[float] = []
    jitter_ms: List[float] = []
    parse_times_ms: List[float] = []
    decompress_times_ms: List[float] = []
    render_times_ms: List[float] = []
    total_client_times_ms: List[float] = []
    robot_latencies_ms: List[float] = []
    payload_sizes_bytes: List[int] = []
    time_steps: List[int] = []

    target_interval_ms = 1000.0 / target_hz
    t_start = time.perf_counter()
    t_last_arrival: Optional[float] = None
    sock.settimeout(3.5)

    try:
        while time.perf_counter() - t_start < duration_s:
            # 1. Read WebSocket Frame Header
            t_recv = None
            while len(buf) < 2:
                chunk = sock.recv(4096)
                if not chunk:
                    break
                buf += chunk
                t_recv = time.perf_counter()
            if len(buf) < 2:
                break

            b0, b1 = buf[0], buf[1]
            opcode = b0 & 0x0F
            plen = b1 & 0x7F
            idx = 2

            # Close frame received
            if opcode == 0x08:
                break
            # Ping frame -> reply Pong
            elif opcode == 0x09:
                pong = bytearray(buf[:idx])
                pong[0] = (pong[0] & 0xF0) | 0x0A
                sock.sendall(pong)
                buf = buf[idx:]
                continue

            if plen == 126:
                while len(buf) < 4:
                    chunk = sock.recv(4096)
                    if not chunk: break
                    buf += chunk
                    t_recv = time.perf_counter()
                plen = struct.unpack("!H", buf[2:4])[0]
                idx = 4
            elif plen == 127:
                while len(buf) < 10:
                    chunk = sock.recv(4096)
                    if not chunk: break
                    buf += chunk
                    t_recv = time.perf_counter()
                plen = struct.unpack("!Q", buf[2:10])[0]
                idx = 10

            while len(buf) < idx + plen:
                chunk = sock.recv(idx + plen - len(buf))
                if not chunk: break
                buf += chunk
                t_recv = time.perf_counter()

            payload = buf[idx : idx + plen]
            buf = buf[idx + plen :]

            # Record arrival timestamp
            t_arrival = t_recv if t_recv is not None else time.perf_counter()
            t_arrival_wall = time.time()
            if t_last_arrival is not None:
                dt_iv = (t_arrival - t_last_arrival) * 1000.0
                intervals_ms.append(dt_iv)
                jitter_ms.append(abs(dt_iv - target_interval_ms))
            t_last_arrival = t_arrival
            payload_sizes_bytes.append(len(payload))

            # 2. Emulate Browser Binary Unpacking & Decompression Pipeline
            t_p0 = time.perf_counter()
            decomp_ms = 0.0
            raster_ms = 0.0

            if len(payload) >= 232 and payload[:4] == b"FOLK":
                header_fmt = "<4s I I d 16f 4f 4f 16f 3f 2H 4f 5I"
                fields = struct.unpack(header_fmt, payload[:232])
                step_id = fields[2]
                ts_robot = fields[3]
                time_steps.append(step_id)

                # Approximate clock skew/robot-to-client delta
                dt_r2c = (t_arrival_wall - ts_robot) * 1000.0
                robot_latencies_ms.append(dt_r2c)

                # Unpack compression lengths
                prob_comp_len = fields[-5]
                prob_raw_len = fields[-4]
                haz_comp_len = fields[-3]
                meta_len = fields[-1]

                # Decompress costmaps if present (simulating DecompressionStream)
                t_d0 = time.perf_counter()
                if prob_comp_len > 0 and len(payload) >= 232 + prob_comp_len:
                    slice_prob = payload[232 : 232 + prob_comp_len]
                    if prob_comp_len < prob_raw_len:
                        try:
                            _ = zlib.decompress(slice_prob)
                        except Exception:
                            pass
                if haz_comp_len > 0:
                    slice_haz = payload[232 + prob_comp_len : 232 + prob_comp_len + haz_comp_len]
                    try:
                        _ = zlib.decompress(slice_haz)
                    except Exception:
                        pass
                decomp_ms = (time.perf_counter() - t_d0) * 1000.0
                decompress_times_ms.append(decomp_ms)

                # Parse JSON metadata (trajectories, rays, etc.)
                if meta_len > 0:
                    meta_raw = payload[232 + prob_comp_len + haz_comp_len : 232 + prob_comp_len + haz_comp_len + meta_len]
                    try:
                        _ = json.loads(meta_raw.decode("utf-8"))
                    except Exception:
                        pass

                # 3. Emulate Browser 2D Canvas Rasterization (LUT mapping on costmap)
                t_r0 = time.perf_counter()
                if has_numpy:
                    dummy_prob = np.zeros((95, 125), dtype=np.uint8)
                    _ = lut[dummy_prob]
                raster_ms = (time.perf_counter() - t_r0) * 1000.0
                render_times_ms.append(raster_ms)

            parse_ms = (time.perf_counter() - t_p0) * 1000.0
            parse_times_ms.append(parse_ms)
            total_client_times_ms.append(parse_ms + raster_ms)

    except Exception as e:
        if verbose:
            print(f"[WebSocket Warning] Streaming ended: {e}")
    finally:
        sock.close()

    actual_duration_s = time.perf_counter() - t_start
    total_frames = len(payload_sizes_bytes)
    effective_hz = total_frames / max(0.001, actual_duration_s)

    # Frame drops (gaps in sequence IDs)
    drops = 0
    if len(time_steps) > 1:
        for i in range(1, len(time_steps)):
            gap = time_steps[i] - time_steps[i - 1]
            if gap > 3:  # 20 Hz decimation steps by 3
                drops += (gap // 3) - 1

    stutters = sum(1 for iv in intervals_ms if iv > 66.7)
    hitches = sum(1 for iv in intervals_ms if iv > 100.0)

    return {
        "connected": True,
        "mode": "websocket",
        "duration_s": actual_duration_s,
        "total_frames": total_frames,
        "effective_hz": effective_hz,
        "target_hz": target_hz,
        "drops": drops,
        "drop_rate_pct": (drops / max(1, total_frames + drops)) * 100.0,
        "stutters": stutters,
        "stutter_pct": (stutters / max(1, len(intervals_ms))) * 100.0,
        "hitches": hitches,
        "hitch_pct": (hitches / max(1, len(intervals_ms))) * 100.0,
        "intervals": compute_percentiles(intervals_ms),
        "jitter": compute_percentiles(jitter_ms),
        "robot_latencies": compute_percentiles(robot_latencies_ms),
        "parse": compute_percentiles(parse_times_ms),
        "decompress": compute_percentiles(decompress_times_ms),
        "render": compute_percentiles(render_times_ms),
        "total_client": compute_percentiles(total_client_times_ms),
        "payload_kb": (statistics.mean(payload_sizes_bytes) / 1024.0) if payload_sizes_bytes else 0.0,
    }


def benchmark_http_polling(
    base_url: str,
    duration_s: float = 6.0,
    target_hz: float = 20.0,
    verbose: bool = True,
) -> Dict[str, Any]:
    """
    Simulates the browser HTTP polling loop (/api/live_bin) across duration_s,
    measuring network Request-Response Round Trip Time (RTT), interval jitter,
    and client deserialization latency.
    """
    url_bin = f"{base_url}/api/live_bin"
    if verbose:
        print(f"\n[HTTP Profiler] Benchmarking HTTP Polling ({url_bin}) for {duration_s:.1f}s...")

    rtt_ms_list: List[float] = []
    intervals_ms: List[float] = []
    parse_ms_list: List[float] = []
    payload_sizes_bytes: List[int] = []

    target_dt_s = 1.0 / target_hz
    t_start = time.perf_counter()
    t_last_arrival: Optional[float] = None

    while time.perf_counter() - t_start < duration_s:
        t0 = time.perf_counter()
        payload = b""
        try:
            req = urllib.request.Request(url_bin, headers={"User-Agent": "FolkraceLatencyTester/1.0"})
            with urllib.request.urlopen(req, timeout=1.5) as resp:
                payload = resp.read()
            t_resp = time.perf_counter()
            rtt = (t_resp - t0) * 1000.0
            rtt_ms_list.append(rtt)
            payload_sizes_bytes.append(len(payload))

            if t_last_arrival is not None:
                intervals_ms.append((t_resp - t_last_arrival) * 1000.0)
            t_last_arrival = t_resp

            # Parse
            t_p0 = time.perf_counter()
            if len(payload) >= 232 and payload[:4] == b"FOLK":
                meta_len = struct.unpack("<I", payload[228:232])[0]
                if meta_len > 0 and len(payload) >= 232 + meta_len:
                    _ = json.loads(payload[232 : 232 + meta_len].decode("utf-8"))
            parse_ms_list.append((time.perf_counter() - t_p0) * 1000.0)

        except Exception as e:
            time.sleep(0.02)

        # Target interval rate sleep
        elapsed_s = time.perf_counter() - t0
        sleep_s = max(0.0, target_dt_s - elapsed_s)
        time.sleep(sleep_s)

    actual_duration_s = time.perf_counter() - t_start
    total_reqs = len(rtt_ms_list)
    effective_hz = total_reqs / max(0.001, actual_duration_s)

    stutters = sum(1 for iv in intervals_ms if iv > 66.7)
    hitches = sum(1 for iv in intervals_ms if iv > 100.0)

    return {
        "mode": "http_polling",
        "duration_s": actual_duration_s,
        "total_requests": total_reqs,
        "effective_hz": effective_hz,
        "stutters": stutters,
        "stutter_pct": (stutters / max(1, len(intervals_ms))) * 100.0,
        "hitches": hitches,
        "hitch_pct": (hitches / max(1, len(intervals_ms))) * 100.0,
        "rtt": compute_percentiles(rtt_ms_list),
        "intervals": compute_percentiles(intervals_ms),
        "parse": compute_percentiles(parse_ms_list),
        "payload_kb": (statistics.mean(payload_sizes_bytes) / 1024.0) if payload_sizes_bytes else 0.0,
    }

def clamp_params(p: Dict[str, Any]) -> Dict[str, Any]:
    out: Dict[str, Any] = {}
    for k, (lo, hi, is_int) in BOUNDS.items():
        val = max(lo, min(hi, float(p[k])))
        out[k] = int(round(val)) if is_int else round(val, 4)
    return out


def mutate_params(parent: Dict[str, Any], sigma: float, rng: random.Random) -> Dict[str, Any]:
    child: Dict[str, Any] = {}
    for k, (lo, hi, is_int) in BOUNDS.items():
        span = hi - lo
        if is_int:
            if rng.random() < 0.35:
                child[k] = parent[k] + rng.choice([-1, 1])
            else:
                child[k] = parent[k]
        else:
            child[k] = float(parent[k]) + rng.gauss(0.0, sigma * span)
    return clamp_params(child)


def score_results(res_ww: dict, res_tr: dict) -> Tuple[float, int, int, List[str]]:
    all_r = list(res_ww.get("results", [])) + list(res_tr.get("results", []))
    total = len(all_r)
    passed_cnt = sum(1 for r in all_r if r.get("passed"))
    failed_names = [r.get("name", "?") for r in all_r if not r.get("passed")]

    fitness = 1000.0
    det_times: List[float] = []
    rec_times: List[float] = []
    post_dists: List[float] = []
    clean_prog = 0.0

    for r in all_r:
        if r.get("flipped"):
            fitness -= 5000.0
        if r.get("off_track") and r.get("name") != "T2_clean_rev_opponents_25s":
            fitness -= 5000.0
        if "ww_triggers" in r:
            # Clean scenario
            ww = int(r.get("ww_fp", r.get("ww_triggers", 0)))
            fitness -= 10000.0 * ww
            clean_prog += float(r.get("progress_m", 0.0))
            if not r.get("passed"):
                fitness -= 2000.0
        else:
            # Spin scenario
            fitness -= 10000.0 * int(r.get("pre_ww", 0))
            fitness -= 1000.0 * int(r.get("retriggers", 0))
            if not r.get("passed"):
                fitness -= 2000.0
            if r.get("detect_s") is not None:
                det_times.append(float(r["detect_s"]))
            else:
                det_times.append(2.0)
            if r.get("rec_done_s") is not None:
                rec_times.append(float(r["rec_done_s"]))
            else:
                rec_times.append(4.5)
            post_dists.append(float(r.get("post_rec_dist_m", 0.0)))

    if det_times:
        fitness -= 250.0 * (sum(det_times) / len(det_times))
    if rec_times:
        fitness -= 150.0 * (sum(rec_times) / len(rec_times))
        fitness -= 80.0 * max(rec_times)
    if post_dists:
        fitness += 20.0 * (sum(post_dists) / len(post_dists))
    fitness += 3.0 * clean_prog

    return round(fitness, 2), passed_cnt, total, failed_names


def evaluate_candidate(cand_id: int, params: Dict[str, Any], worker_slot: int) -> Tuple[float, int, int, List[str], Dict[str, Any]]:

        scored.sort(key=lambda x: x[0], reverse=True)
        top = scored[0]
        if best_ever is None or top[0] > best_ever[0]:
            best_ever = top

        print(
            f"[Gen {gen+1}/{args.gens}] Best={top[0]:.2f} ({top[1]}/{top[2]} PASS, fail={top[3]}) "
            f"| EverBest={best_ever[0]:.2f} ({best_ever[1]}/{best_ever[2]} PASS) "
            f"| {time.perf_counter()-t_gen:.1f}s",
            flush=True,
        )

        # Elitist selection (top 2) + mutated offspring
        elites = [scored[0][4], scored[min(1, len(scored) - 1)][4]]
        sigma = max(0.08, 0.22 * (0.75 ** (gen + 1)))
        next_pop = [dict(elites[0])]
        while len(next_pop) < args.pop:
            parent = rng.choice(elites)
            next_pop.append(mutate_params(parent, sigma=sigma, rng=rng))
        population = next_pop

    print("\n" + "=" * 80)
    print(f"  OPTIMIZATION COMPLETE in {time.perf_counter()-t_start:.1f}s")
    print(f"  Best Fitness: {best_ever[0]:.2f} ({best_ever[1]}/{best_ever[2]} PASS, failed={best_ever[3]})")
    print("  Best Params JSON:")
    print(json.dumps(best_ever[4], indent=2))
    print("=" * 80)
    return 0

    raise SystemExit(main())

#!/usr/bin/env python3
"""Run a high-rate STM32 FOC trace while the ESP32 commands the motors."""

from __future__ import annotations

import argparse
import json
import math
import statistics
import struct
import time
from pathlib import Path

import serial


TRACE_HEADER = struct.Struct("<2sHHHII")
TRACE_SAMPLE = struct.Struct("<HHhhhhhh")
TRACE_SYNC = b"FT"
MOTOR_POLE_PAIRS = 11
RAW_COUNTS = 16384


def read_trace(port: serial.Serial) -> tuple[dict[str, int], list[tuple[int, ...]]]:
    data = bytearray()
    deadline = time.monotonic() + 8.0
    expected = None
    while time.monotonic() < deadline:
        chunk = port.read(4096)
        if chunk:
            data.extend(chunk)
        start = data.find(TRACE_SYNC)
        if start >= 0 and len(data) >= start + TRACE_HEADER.size:
            header = TRACE_HEADER.unpack_from(data, start)
            count = header[3]
            expected = start + TRACE_HEADER.size + count * TRACE_SAMPLE.size
            if len(data) >= expected:
                payload = data[start + TRACE_HEADER.size : expected]
                samples = [
                    TRACE_SAMPLE.unpack_from(payload, offset)
                    for offset in range(0, len(payload), TRACE_SAMPLE.size)
                ]
                return (
                    {
                        "version": header[1],
                        "sample_hz": header[2],
                        "sample_count": count,
                        "overwritten": header[4],
                    },
                    samples,
                )
    raise TimeoutError(f"trace dump incomplete; expected_end={expected}, received={len(data)}")


def unwrap_delta(delta: int) -> int:
    half = RAW_COUNTS // 2
    return ((delta + half) % RAW_COUNTS) - half


def motor_analysis(samples: list[tuple[int, ...]], motor_index: int) -> dict[str, object]:
    raw_index = motor_index
    target_index = 2 + motor_index
    actual_index = 4 + motor_index
    uq_index = 6 + motor_index
    raw = [sample[raw_index] for sample in samples]
    deltas = [unwrap_delta(b - a) for a, b in zip(raw, raw[1:])]
    electrical_steps = [
        delta * (2.0 * math.pi * MOTOR_POLE_PAIRS / RAW_COUNTS) for delta in deltas
    ]

    actual = [sample[actual_index] for sample in samples]
    target = [sample[target_index] for sample in samples]
    uq = [sample[uq_index] for sample in samples]
    actual_steps = [abs(b - a) for a, b in zip(actual, actual[1:])]
    phase_jumps = [abs(step) for step in electrical_steps]
    phase_tracking_errors = [
        measured - (
            target[index]
            * (2.0 * math.pi / 60.0)
            * MOTOR_POLE_PAIRS
            / 500.0
        )
        for index, measured in enumerate(electrical_steps)
    ]
    return {
        "target_rpm_avg": round(statistics.mean(target), 1),
        "actual_rpm_avg": round(statistics.mean(actual), 1),
        "actual_rpm_minmax": [min(actual), max(actual)],
        "uq_mv_max_abs": max(abs(value) for value in uq),
        "max_raw_delta_counts": max(abs(value) for value in deltas),
        "max_electrical_step_rad": round(max(phase_jumps), 3),
        "electrical_steps_over_pi": sum(value > math.pi for value in phase_jumps),
        "max_phase_tracking_error_rad": round(max(abs(value) for value in phase_tracking_errors), 3),
        "phase_tracking_errors_over_1rad": sum(
            abs(value) > 1.0 for value in phase_tracking_errors
        ),
        "actual_steps_over_40rpm": sum(value > 40 for value in actual_steps),
    }


def main() -> None:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--stm-port", default="COM13")
    parser.add_argument("--esp-port", default="COM5")
    parser.add_argument("--duration", type=float, default=10.0)
    parser.add_argument("--speed", type=float, default=-0.8)
    parser.add_argument("--steer-angle", type=int, default=90)
    parser.add_argument("--output", type=Path, default=Path("stm32_foc_trace.bin"))
    args = parser.parse_args()

    with serial.Serial(args.stm_port, 921600, timeout=0.01) as stm, serial.Serial(
        args.esp_port, 921600, timeout=0.005
    ) as esp:
        stm.reset_input_buffer()
        esp.reset_input_buffer()
        stm.write(b"TRACE_START\n")
        stm.flush()
        time.sleep(0.1)
        try:
            for command in (
                b"MOTORS\n",
                f"A{args.steer_angle}\n".encode(),
                f"V {args.speed}\n".encode(),
            ):
                esp.write(command)
                esp.flush()
                time.sleep(0.15)
            time.sleep(args.duration)
        finally:
            esp.write(b"ESTOP\n")
            esp.flush()
            time.sleep(0.5)
            stm.write(b"TRACE_STOP\n")
            stm.flush()
            time.sleep(0.1)
            stm.reset_input_buffer()
            stm.write(b"TRACE_DUMP\n")
            stm.flush()
        header, samples = read_trace(stm)

    raw = b"".join(TRACE_SAMPLE.pack(*sample) for sample in samples)
    args.output.write_bytes(TRACE_HEADER.pack(TRACE_SYNC, header["version"], header["sample_hz"], header["sample_count"], header["overwritten"], 0) + raw)
    result = {
        "trace_file": str(args.output),
        "header": header,
        "motor1": motor_analysis(samples, 0),
        "motor2": motor_analysis(samples, 1),
    }
    print(json.dumps(result, indent=2))


if __name__ == "__main__":
    main()

def compute_derived_params(cfg: dict) -> dict:
    v = cfg["vehicle"]
    d = cfg["dynamics"]
    t = cfg["timing"]

    rear_offset = float(v["rear_axle_offset_m"])
    front_offset = float(v["front_axle_offset_m"])
    wheelbase = front_offset - rear_offset

    tire_mu = float(d["tire_mu"])
    gravity_g = float(d["gravity_g"])
    a_max_base = tire_mu * gravity_g

    accel_fwd_max = float(d["accel_fwd_ratio"]) * a_max_base
    brake_decel_max = float(d["brake_decel_ratio"]) * a_max_base
    lat_accel_max = float(d["lat_accel_ratio"]) * a_max_base

    max_steer_rad = float(v["max_steering_rad"])
    r_min_turn = wheelbase / math.tan(max_steer_rad)
    v_min_cruise = math.sqrt(lat_accel_max * r_min_turn)

    ctrl_freq = float(t["control_loop_freq_hz"])
    ctrl_period_us = int(1_000_000 / ctrl_freq)
    ctrl_dt_s = 1.0 / ctrl_freq

    return {
        "wheelbase_m": wheelbase,
        "a_max_base": a_max_base,
        "accel_fwd_max": accel_fwd_max,
        "brake_decel_max": brake_decel_max,
        "lat_accel_max": lat_accel_max,
        "r_min_turn": r_min_turn,
        "v_min_cruise": v_min_cruise,
        "ctrl_period_us": ctrl_period_us,
        "ctrl_dt_s": ctrl_dt_s,
    }

def sync_all():
    cfg = load_config()
    derived = compute_derived_params(cfg)

    # 1. Generate Python params
    PY_OUT_PATH.parent.mkdir(parents=True, exist_ok=True)
    py_code = generate_python(cfg, derived)
    with open(PY_OUT_PATH, "w", encoding="utf-8") as f:
        f.write(py_code)

    # 2. Generate C++ header
    H_OUT_PATH.parent.mkdir(parents=True, exist_ok=True)
    h_code = generate_c_header(cfg, derived)
    with open(H_OUT_PATH, "w", encoding="utf-8") as f:
        f.write(h_code)

    print(f"[sync_params] Synced vehicle params -> {PY_OUT_PATH.name} and {H_OUT_PATH.name}")


# PlatformIO SCons execution support
try:
    import SCons.Script  # type: ignore
    # Running inside PlatformIO SCons
    sync_all()
except (ImportError, AttributeError):
    # Standalone CLI execution
    if __name__ == "__main__":
        sync_all()
