def ray_segment_intersect(
    ox: float, oy: float, dx: float, dy: float,
    ax: float, ay: float, bx: float, by: float,
) -> Optional[float]:
    """2D ray (o + t*d, t>=0) vs segment (a, b) intersection distance t."""
    sx = bx - ax
    sy = by - ay
    denom = dx * sy - dy * sx
    if abs(denom) < 1e-9:
        return None
    qpx = ax - ox
    qpy = ay - oy
    t = (qpx * sy - qpy * sx) / denom
    u = (qpx * dy - qpy * dx) / denom
    if t >= 0.0 and 0.0 <= u <= 1.0:
        return t
    return None


def segments_from_polyline(pts: np.ndarray, closed: bool = True) -> List[Tuple[float, float, float, float]]:
    segs: List[Tuple[float, float, float, float]] = []
    n = len(pts)
    if n < 2:
        return segs
    for i in range(n - 1):
        segs.append((float(pts[i, 0]), float(pts[i, 1]), float(pts[i + 1, 0]), float(pts[i + 1, 1])))
    if closed:
        segs.append((float(pts[-1, 0]), float(pts[-1, 1]), float(pts[0, 0]), float(pts[0, 1])))
    return segs


def box_segments(cx: float, cy: float, half_w: float, half_l: float) -> List[Tuple[float, float, float, float]]:
    pts = np.array([
        [cx - half_w, cy - half_l],
        [cx + half_w, cy - half_l],
        [cx + half_w, cy + half_l],
        [cx - half_w, cy + half_l],
    ])
    return segments_from_polyline(pts, closed=True)


def point_to_segments_dist(px: float, py: float, segs: List[Tuple[float, float, float, float]]) -> float:
    best_sq = 1e9
    for ax, ay, bx, by in segs:
        vx, vy = bx - ax, by - ay
        l2 = vx * vx + vy * vy
        if l2 < 1e-9:
            d2 = (px - ax) ** 2 + (py - ay) ** 2
        else:
            t = max(0.0, min(1.0, ((px - ax) * vx + (py - ay) * vy) / l2))
            qx, qy = ax + t * vx, ay + t * vy
            d2 = (px - qx) ** 2 + (py - qy) ** 2
        if d2 < best_sq:
            best_sq = d2
    return math.sqrt(best_sq)


def raycast_16_tofs(
    x: float,
    y: float,
    yaw: float,
    segs: List[Tuple[float, float, float, float]],
) -> Dict[str, float]:
    """Simulates all 16 VL53L8CX beams in world frame (+Y forward, +X right, +yaw CCW)."""
    cos_y = math.cos(yaw)
    sin_y = math.sin(yaw)
    readings: Dict[str, float] = {}
    for s_name, sx_loc, sy_loc, h_loc, max_r in FRONT_TOF_SENSOR_RIG:
        # Local sensor origin -> world
        ox = x + cos_y * sx_loc - sin_y * sy_loc
        oy = y + sin_y * sx_loc + cos_y * sy_loc
        # Local sensor ray dir: (cos(h_loc), sin(h_loc)) in (+X right, +Y fwd)
        dlx = math.cos(h_loc)
        dly = math.sin(h_loc)
        dx = cos_y * dlx - sin_y * dly
        dy = sin_y * dlx + cos_y * dly

        best_t = max_r
        for ax, ay, bx, by in segs:
            t = ray_segment_intersect(ox, oy, dx, dy, ax, ay, bx, by)
            if t is not None and t < best_t:
                best_t = t
        readings[s_name] = max(0.030, min(TOF_MAX_RANGE_M, best_t))
    return readings


class SimCar2D:
    """Simple 60 Hz Ackermann bicycle simulator with tire grip limit & wall contact."""

    def __init__(self, x: float = 0.0, y: float = 0.0, yaw: float = 0.0):
        self.x = x
        self.y = y
        self.yaw = yaw
        self.v = 0.0
        self.steer = 0.0
        self.yaw_rate = 0.0
        self.wall_hits = 0

    def step(
        self,
        v_cmd: float,
        steer_cmd: float,
        dt: float,
        segs: List[Tuple[float, float, float, float]],
        force_stall: bool = False,
    ) -> None:
        max_ds = SERVO_MAX_SPEED_RAD_S * dt
        self.steer += max(-max_ds, min(max_ds, steer_cmd - self.steer))
        self.steer = max(-MAX_STEERING_RAD, min(MAX_STEERING_RAD, self.steer))

        max_dv = TIRE_MU * GRAVITY_G * dt
        self.v += max(-max_dv, min(max_dv, v_cmd - self.v))

        if force_stall:
            self.v = 0.0
            self.yaw_rate = 0.0
            return

        # Kinematic yaw rate with friction-circle understeer cap
        raw_omega = (self.v / WHEELBASE) * math.tan(self.steer)
        max_omega = (TIRE_MU * GRAVITY_G) / max(0.5, abs(self.v))
        self.yaw_rate = max(-max_omega, min(max_omega, raw_omega))

        yaw_mid = self.yaw + 0.5 * self.yaw_rate * dt
        nx = self.x - self.v * math.sin(yaw_mid) * dt
        ny = self.y + self.v * math.cos(yaw_mid) * dt
        nyaw = wrap_to_pi(self.yaw + self.yaw_rate * dt)

        # Wall contact clamp (chassis half-width ~ 0.075 m)
        d_wall = point_to_segments_dist(nx, ny, segs)
        if d_wall < 0.070:
            self.wall_hits += 1
            self.v *= 0.15
            self.yaw_rate *= 0.3
        else:
            self.x = nx
            self.y = ny
            self.yaw = nyaw


def make_corridor_walls(width: float = 1.0, length: float = 10.0) -> List[Tuple[float, float, float, float]]:
    hw = width * 0.5
    return [
        (-hw, -2.0, -hw, length),
        (hw, -2.0, hw, length),
        (-hw, -2.0, hw, -2.0),
        (-hw, length, hw, length),
    ]


def make_hairpin_walls() -> List[Tuple[float, float, float, float]]:
    """Closed oval hairpin: 1.0m left lane [-0.5, +0.5], dividing wall at x=0.50..0.60, 1.0m right lane [0.60, 1.60]."""
    return [
        (-0.50, -1.5, -0.50, 5.8),
        (-0.50, 5.8, 1.60, 5.8),
        (1.60, 5.8, 1.60, -1.5),
        (1.60, -1.5, -0.50, -1.5),
        (0.50, -0.3, 0.50, 4.6),
        (0.60, -0.3, 0.60, 4.6),
        (0.50, 4.6, 0.60, 4.6),
        (0.50, -0.3, 0.60, -0.3),
    ]


def run_step(brain: AutonomousBrain, car: SimCar2D, segs: List[Tuple[float, float, float, float]], sim_t: float, dt: float, force_stall: bool = False):
    tofs = raycast_16_tofs(car.x, car.y, car.yaw, segs)
    v_cmd, steer_cmd, telem = brain.step(
        tof_readings=tofs,
        v_est=car.v,
        steer_actual_rad=car.steer,
        gyro_z_rad_s=car.yaw_rate,
        dt=dt,
        sim_time_s=sim_t,
        ground_truth_pose=None,
        full_telemetry=True,
        imu_yaw_rad=car.yaw,
    )
    car.step(v_cmd, steer_cmd, dt, segs, force_stall=force_stall)
    return telem


def scenario_s1_straight_180_spin() -> Tuple[bool, str]:
    brain = AutonomousBrain()
    brain.reset((0.0, 0.0, -math.pi / 2.0))
    car = SimCar2D(0.0, 0.0, -math.pi / 2.0)
    segs = make_corridor_walls(width=1.0, length=40.0)
    dt = 1.0 / 60.0
    sim_t = 0.0

    # Drive forward 2.5 s
    for _ in range(150):
        sim_t += dt
        run_step(brain, car, segs, sim_t, dt)

    # Inject 180 deg spinout
    car.yaw = math.pi
    car.v = 0.1
    brain.pose_tracker.yaw = math.pi
    brain.pose_tracker._imu_yaw_offset = 0.0

    t_spin = sim_t
    detect_lat = None
    rec_done = None
    aligned_at_exit = False
    retriggers = 0
    was_in_180 = False

    for _ in range(300):  # 5.0 s max
        sim_t += dt
        telem = run_step(brain, car, segs, sim_t, dt)
        is_ww = bool(telem.get("is_wrong_way"))
        rec_state = telem.get("recovery_state", "")
        in_180 = (rec_state == StuckRecoveryState.CORRECTIVE_180)

        if is_ww and detect_lat is None:
            detect_lat = sim_t - t_spin
        if rec_done is None and was_in_180 and not in_180:
            rec_done = sim_t - t_spin
            aligned_at_exit = abs(wrap_to_pi(car.yaw - 0.0)) <= math.radians(35.0)
        elif rec_done is not None and is_ww:
            retriggers += 1
        was_in_180 = in_180

    aligned = abs(wrap_to_pi(car.yaw - 0.0)) <= math.radians(35.0)
    ok = (
        detect_lat is not None and detect_lat <= 0.35
        and rec_done is not None and rec_done <= 2.5
        and aligned_at_exit and aligned and retriggers == 0
    )
    return ok, f"detect={detect_lat:.3f}s rec={rec_done:.3f}s aligned={aligned} retriggers={retriggers} hits={car.wall_hits}"

def scenario_s5_opponent_occlusion_spin() -> Tuple[bool, str]:
    brain = AutonomousBrain()
    brain.reset((0.0, 0.0, -math.pi / 2.0))
    car = SimCar2D(0.0, 0.0, -math.pi / 2.0)
    base_segs = make_corridor_walls(width=1.0, length=40.0)
    dt = 1.0 / 60.0
    sim_t = 0.0
    for _ in range(150):
        sim_t += dt
        run_step(brain, car, base_segs, sim_t, dt)

    # Spin 180 deg and place opponent box 0.80m in front (south of car) for 1.5s
    car.yaw = math.pi
    car.v = 0.0
    brain.pose_tracker.yaw = car.yaw
    brain.pose_tracker._imu_yaw_offset = 0.0
    occ_segs = base_segs + box_segments(car.x, car.y - 0.80, half_w=0.22, half_l=0.10)

    detected_during = False
    for _ in range(90):  # 1.5 s occluded
        sim_t += dt
        telem = run_step(brain, car, occ_segs, sim_t, dt)
        if telem.get("is_wrong_way"):
            detected_during = True

    # Opponent moves away -> trail must still be intact and trigger recovery if not already aligned
    detected_after = False
    for _ in range(210):
        sim_t += dt
        telem = run_step(brain, car, base_segs, sim_t, dt)
        if telem.get("is_wrong_way"):
            detected_after = True

    aligned = abs(wrap_to_pi(car.yaw - 0.0)) <= math.radians(35.0)
    ok = (detected_during or detected_after) and aligned
    return ok, f"det_during={detected_during} det_after={detected_after} aligned={aligned} final_yaw_deg={math.degrees(car.yaw):.1f}"


def scenario_s6_corner_catch_stall() -> Tuple[bool, str]:
    brain = AutonomousBrain()
    brain.reset((0.0, 0.0, -math.pi / 2.0))
    car = SimCar2D(0.0, 0.0, -math.pi / 2.0)
    segs = make_corridor_walls(width=1.0, length=40.0)
    dt = 1.0 / 60.0
    sim_t = 0.0
    for _ in range(150):
        sim_t += dt
        run_step(brain, car, segs, sim_t, dt)

    car.yaw = math.pi
    car.v = 0.0
    brain.pose_tracker.yaw = car.yaw
    brain.pose_tracker._imu_yaw_offset = 0.0

    # Wait for CORRECTIVE_180 to engage, then force stall for 0.6 s
    engaged_rev = False
    stalled_frames = 0
    for _ in range(300):
        sim_t += dt
        in_180_fwd = (
            brain.recovery_manager.state == StuckRecoveryState.CORRECTIVE_180
            and brain.recovery_manager.corrective_substate == "FORWARD"
        )
        force = False
        if in_180_fwd and stalled_frames < 36:
            force = True
            stalled_frames += 1
        telem = run_step(brain, car, segs, sim_t, dt, force_stall=force)
        if brain.recovery_manager.state == StuckRecoveryState.CORRECTIVE_180 and brain.recovery_manager.corrective_substate == "REVERSE":
            engaged_rev = True

    aligned = abs(wrap_to_pi(car.yaw - 0.0)) <= math.radians(35.0)
    ok = engaged_rev and aligned
    return ok, f"engaged_rev={engaged_rev} aligned={aligned} final_yaw_deg={math.degrees(car.yaw):.1f}"


def scenario_s7_midturn_early_exit() -> Tuple[bool, str]:
    brain = AutonomousBrain()
    brain.reset((0.0, 0.0, 0.0))
    for i in range(80):
        brain.watchdog.update(0.0, i * 0.05, 0.0, v_est=2.0, yaw_rate=0.0, now_s=i * 0.05, dt=0.05)

    # Mid-turn at yaw = +pi/2 (facing -X):
    # - gap (-1.5, 0.01) points south (-Y) along old trail -> MUST reject
    # - gap (+1.5, 0.01) points north (+Y) forward -> MUST accept
    ok_bad, info_bad = brain.watchdog.is_gap_early_exit(0.0, 4.0, math.pi / 2.0, -1.5, 0.01)
    ok_good, info_good = brain.watchdog.is_gap_early_exit(0.0, 4.0, math.pi / 2.0, 1.5, 0.01)
    ok = (not ok_bad) and ok_good
    return ok, f"wrong_way_gap_exit={ok_bad} ({info_bad['reason']}), forward_gap_exit={ok_good} ({info_good['reason']})"


def scenario_s8_track_racing_2d_clean_lap() -> Tuple[bool, str]:
    left_pts, right_pts = load_track_walls()
    if len(left_pts) == 0 or len(right_pts) == 0:
        return False, "Failed to load track_racing.wbt walls"
    segs = segments_from_polyline(left_pts, closed=True) + segments_from_polyline(right_pts, closed=True)

    brain = AutonomousBrain()
    brain.reset((0.0, 0.0, -math.pi / 2.0))
    car = SimCar2D(0.0, 0.0, -math.pi / 2.0)
    dt = 1.0 / 60.0
    sim_t = 0.0
    ww_triggers = 0
    dist_traveled = 0.0
    px, py = car.x, car.y

    for _ in range(900):  # 15.0 s
        sim_t += dt
        telem = run_step(brain, car, segs, sim_t, dt)
        dist_traveled += math.hypot(car.x - px, car.y - py)
        px, py = car.x, car.y
        if telem.get("is_wrong_way"):
            ww_triggers += 1

    ok = ww_triggers == 0 and dist_traveled >= 15.0
    return ok, f"ww_triggers={ww_triggers} dist={dist_traveled:.1f}m hits={car.wall_hits}"

    scenarios = [
        ("S1_straight_180_spin", scenario_s1_straight_180_spin),
        ("S2_hairpin_no_fp", scenario_s2_hairpin_no_fp),
        ("S3_wall_hugging_3pt", scenario_s3_wall_hugging_3pt),
        ("S4_shoved_forward_spin", scenario_s4_shoved_forward_spin),
        ("S5_opponent_occlusion_spin", scenario_s5_opponent_occlusion_spin),
        ("S6_corner_catch_stall", scenario_s6_corner_catch_stall),
        ("S7_midturn_early_exit", scenario_s7_midturn_early_exit),
        ("S8_track_racing_2d_clean_lap", scenario_s8_track_racing_2d_clean_lap),
    ]

    print("=" * 80)
    print("  2D CLOSED-LOOP WRONG-WAY & RECOVERY SIMULATION BENCH")
    print("=" * 80)
    t0 = time.perf_counter()
    all_ok = True
    for name, fn in scenarios:
        passed, detail = fn()
        tag = "PASS" if passed else "FAIL"
        print(f"  [{tag}] {name:30s} | {detail}")
        if not passed:
            all_ok = False
    print(f"Finished in {time.perf_counter() - t0:.2f}s")
    print("=" * 80)
    return 0 if all_ok else 1


if __name__ == "__main__":
    raise SystemExit(main())

        self.send_header("Content-Length", str(len(raw)))
        self.end_headers()
        self.wfile.write(raw)

    def _serve_static_file(self, filename: str, content_type: str) -> None:
        filepath = os.path.join(STATIC_DIR, filename)
        with open(filepath, "rb") as f:
            data = f.read()
        self.send_response(200)
        self.send_header("Content-Type", content_type)
        self.send_header("Content-Length", str(len(data)))
        self.end_headers()
        self.wfile.write(data)

    def do_GET(self):
        parsed = urllib.parse.urlparse(self.path)
        if parsed.path in ("/", "/index.html"):
            self._serve_static_file("track_builder.html", "text/html; charset=utf-8")
            return
        if parsed.path == "/track_builder.css":
            self._serve_static_file("track_builder.css", "text/css; charset=utf-8")
            return
        if parsed.path == "/track_builder.js":
            self._serve_static_file("track_builder.js", "application/javascript; charset=utf-8")
            return

        return super().do_GET()

    def do_POST(self):
        parsed = urllib.parse.urlparse(self.path)
        if parsed.path not in ("/api/save_local", "/api/deploy_to_pi"):
            self._send_json(404, {"status": "error", "message": "Endpoint not found"})
            return

        try:
            content_length = int(self.headers.get("Content-Length", 0))
            body = self.rfile.read(content_length)
            data = json.loads(body.decode("utf-8"))
        except (ValueError, UnicodeDecodeError) as e:
            self._send_json(400, {"status": "error", "message": f"Invalid JSON payload: {e}"})
            return

        try:
            output_dir = self.server.output_dir
            os.makedirs(output_dir, exist_ok=True)
            saved_files = []

            png_b64 = data.get("png_b64", "")
            if png_b64:
                try:
                    png_bytes = base64.b64decode(png_b64, validate=True)
                except Exception as e:
                    self._send_json(400, {"status": "error", "message": f"Invalid base64 PNG data: {e}"})
                    return
                with open(os.path.join(output_dir, "track_map.png"), "wb") as f:
                    f.write(png_bytes)
                saved_files.append("track_map.png")

            yaml_content = data.get("yaml", "")
            if yaml_content:
                with open(os.path.join(output_dir, "track_map.yaml"), "w", encoding="utf-8") as f:
                    f.write(yaml_content)
                saved_files.append("track_map.yaml")

            track_json = data.get("track_json", "")
            if track_json:
                with open(os.path.join(output_dir, "track_layout.json"), "w", encoding="utf-8") as f:
                    f.write(track_json)
                saved_files.append("track_layout.json")

            wbt_content = data.get("wbt", "")
            if wbt_content:
                with open(os.path.join(output_dir, "track_world.wbt"), "w", encoding="utf-8") as f:
                    f.write(wbt_content)
                saved_files.append("track_world.wbt")

            try:
                display_dir = os.path.relpath(output_dir, ROOT_DIR)
            except ValueError:
                display_dir = output_dir

            print(f"[OK] Saved track bundle ({', '.join(saved_files)}) in: {output_dir}")
            self._send_json(
                200,
                {
                    "status": "ok",
                    "message": f"Saved track bundle to {display_dir}",
                    "files": saved_files,
                },
            )
        except Exception as e:
            self._send_json(500, {"status": "error", "message": str(e)})

def _kill_proc(proc):
    if proc is None or proc.poll() is not None:
        return
    try:

                ["netsh", "wlan", "show", "interfaces"],
                stdout=subprocess.PIPE,
                stderr=subprocess.DEVNULL,
                text=True,
                timeout=2,
            )
            if res.returncode == 0:
                in_connected = False
                for line in res.stdout.splitlines():
                    if ":" not in line:
                        continue
                    key, val = [p.strip() for p in line.split(":", 1)]
                    if key == "Name":
                        iface_name = val
                        in_connected = False
                    elif key == "State" and val.lower() == "connected":
                        in_connected = True
                    elif key == "SSID" and in_connected and val:
                        ssid = val
                        break
        except Exception:
            pass

        if iface_name and ssid:
            try:
                res = subprocess.run(
                    ["netsh", "interface", "ip", "show", "addresses", iface_name],
                    stdout=subprocess.PIPE,
                    stderr=subprocess.DEVNULL,
                    text=True,
                    timeout=2,
                )
                if res.returncode == 0:
                    m = re.search(r"IP Address:\s*(\d+\.\d+\.\d+\.\d+)", res.stdout)
                    if m:
                        local_ip = m.group(1)
            except Exception:
                pass

    if not local_ip:
        try:
            with socket.socket(socket.AF_INET, socket.SOCK_DGRAM) as s:
                s.connect(("8.8.8.8", 80))
                local_ip = s.getsockname()[0]
        except Exception:
            pass

    subnet_prefix = ".".join(local_ip.split(".")[:3]) + "." if local_ip and local_ip.count(".") == 3 else None
    return ssid, local_ip, subnet_prefix


def discover_arp_pi_ips(subnet_prefix: Optional[str] = None) -> List[str]:
    """Reads the local OS ARP cache for Raspberry Pi MAC prefixes on the active subnet."""
    ips: List[str] = []
    try:
        res = subprocess.run(["arp", "-a"], stdout=subprocess.PIPE, stderr=subprocess.DEVNULL, text=True, timeout=2)
        if res.returncode != 0:
            return ips
        for line in res.stdout.splitlines():
            m = re.search(r"(\d+\.\d+\.\d+\.\d+)\s+([0-9a-fA-F]{2}(?:[:-][0-9a-fA-F]{2}){5})", line)
            if not m:
                continue
            ip, mac = m.group(1), m.group(2).lower().replace("-", ":")
            if subnet_prefix and not ip.startswith(subnet_prefix):
                continue
            if mac.startswith(PI_MAC_PREFIXES) and ip not in ips:
                ips.append(ip)
    except Exception:
        pass
    return ips


def _is_tcp_open(ip: str, port: int, timeout_s: float = 0.35) -> bool:
    try:
        with socket.create_connection((ip, port), timeout=timeout_s):
            return True
    except Exception:
        return False


def scan_wifi_subnet_for_pi(
    subnet_prefix: str, local_ip: Optional[str], port: int, timeout_s: float = 0.8
) -> List[str]:
    """Probes the current Wi-Fi /24 subnet in parallel for hosts listening on `port`."""
    targets = [f"{subnet_prefix}{i}" for i in range(1, 255) if f"{subnet_prefix}{i}" != local_ip]
    open_ips: List[str] = []
    with concurrent.futures.ThreadPoolExecutor(max_workers=64) as pool:
        futures = {pool.submit(_is_tcp_open, ip, port, timeout_s): ip for ip in targets}
        for fut in concurrent.futures.as_completed(futures):
            ip = futures[fut]
            try:
                if fut.result():
                    open_ips.append(ip)
            except Exception:
                pass
    # Prioritize hosts whose MAC in the refreshed ARP cache matches a Raspberry Pi OUI
    arp_pis = set(discover_arp_pi_ips(subnet_prefix))
    open_ips.sort(key=lambda ip: (0 if ip in arp_pis else 1, ip))
    return open_ips


def get_candidate_hosts(preferred_host: Optional[str] = None, subnet_prefix: Optional[str] = None) -> List[str]:
    """Returns deduplicated candidate hosts restricted to the active Wi-Fi subnet."""
    if preferred_host:
        return [preferred_host]
    if subnet_prefix is None:
        _, _, subnet_prefix = get_active_wifi_network()

    ordered: List[str] = []
    for ip in discover_arp_pi_ips(subnet_prefix):
        if ip not in ordered:
            ordered.append(ip)

    for ip in DEFAULT_WIFI_IPS:
        if (subnet_prefix is None or ip.startswith(subnet_prefix)) and ip not in ordered:
            ordered.append(ip)

    if subnet_prefix is None or subnet_prefix == "192.168.20.":
        if "robot" not in ordered:
            ordered.append("robot")

    if "robot.local" not in ordered:
        ordered.append("robot.local")
    return ordered


def format_ssh_target(host: Optional[str]) -> str:
    """Formats a hostname or IP for SSH/SCP (uses 'robot' alias or 'admin@<ip>')."""
    if not host or host in ("robot", "robot.local", "127.0.0.1"):
        return "robot"
    if "@" in host:
        return host
    return f"admin@{host}"


def _probe_http_status(host: str, timeout_s: float = 1.0) -> Optional[str]:
    url = f"http://{host}:{PORT}/api/status"
    try:
        req = urllib.request.Request(url, headers={"User-Agent": "FolkraceLauncher/1.0"})
        with urllib.request.urlopen(req, timeout=timeout_s) as resp:
            if resp.status == 200:
                return f"http://{host}:{PORT}"
    except Exception:
        pass
    return None


def resolve_pi_host(preferred_host: Optional[str] = None, timeout_s: float = 1.0) -> Tuple[Optional[str], Optional[str]]:
    """
    Auto-detects the responsive hostname or IP for the Pi 4 debug server on the active Wi-Fi.
    Returns (hostname, base_url).
    """
    _, local_ip, subnet_prefix = get_active_wifi_network()
    candidates = get_candidate_hosts(preferred_host, subnet_prefix=subnet_prefix)

    for host in candidates:
        base_url = _probe_http_status(host, timeout_s=timeout_s)
        if base_url:
            return host, base_url

    if not preferred_host and subnet_prefix:
        for ip in scan_wifi_subnet_for_pi(subnet_prefix, local_ip, PORT):
            base_url = _probe_http_status(ip, timeout_s=timeout_s)
            if base_url:
                return ip, base_url

    return None, None


def check_ssh_connection(host: str = "robot", timeout_s: int = 3) -> bool:
    """Checks if Raspberry Pi is reachable over SSH."""
    cmd = ["ssh", "-o", "BatchMode=yes", "-o", f"ConnectTimeout={timeout_s}", host, "echo ok"]
    try:
        res = subprocess.run(cmd, stdout=subprocess.PIPE, stderr=subprocess.PIPE, timeout=timeout_s + 1, text=True)
        return res.returncode == 0 and "ok" in res.stdout
    except Exception:
        return False


def resolve_ssh_host(preferred_host: Optional[str] = None, timeout_s: int = 3) -> str:
    """Finds the first reachable SSH target on the active Wi-Fi (including dynamic hotspot IPs)."""
    if preferred_host:
        return format_ssh_target(preferred_host)

    _, local_ip, subnet_prefix = get_active_wifi_network()
    candidates = get_candidate_hosts(subnet_prefix=subnet_prefix)
    seen = set()

    def _try_hosts(hosts: List[str]) -> Optional[str]:
        for host in hosts:
            if host in ("127.0.0.1", "robot.local"):
                continue
            target = format_ssh_target(host)
            if target in seen:
                continue
            seen.add(target)
            probe_ip = "192.168.20.120" if host == "robot" else host
            if not _is_tcp_open(probe_ip, 22, timeout_s=0.4):
                continue
            if check_ssh_connection(target, timeout_s=timeout_s):
                return target
        return None

    found = _try_hosts(candidates)
    if found:
        return found

    if subnet_prefix:
        scanned = scan_wifi_subnet_for_pi(subnet_prefix, local_ip, 22)
        found = _try_hosts(scanned)
        if found:
            return found

    return "robot"


def query_api_json(base_url: str, endpoint: str, timeout_s: float = 3.0) -> Optional[Dict[str, Any]]:
    """Helper to query JSON from the Pi debug server."""
    url = f"{base_url}{endpoint}"
    try:
        req = urllib.request.Request(url, headers={"User-Agent": "FolkraceLauncher/1.0"})
        with urllib.request.urlopen(req, timeout=timeout_s) as resp:
            if resp.status == 200:
                return json.loads(resp.read().decode("utf-8"))
    except Exception as e:
        pass
    return None

def launch_tactical_gui(base_url: str, prefer_app_window: bool = True):
    """
    Launches the Tactical Mission Control GUI (same as browser server).
    If prefer_app_window is True, attempts to launch Edge or Chrome in standalone
    app window mode (--app=http://...) for a borderless native application window
    without browser tabs or URL bars. Falls back to default web browser.
    """
    if prefer_app_window:
        candidates = [
            r"C:\Program Files (x86)\Microsoft\Edge\Application\msedge.exe",
            r"C:\Program Files\Microsoft\Edge\Application\msedge.exe",
            r"C:\Program Files\Google\Chrome\Application\chrome.exe",
            r"C:\Program Files (x86)\Google\Chrome\Application\chrome.exe",
            shutil.which("msedge") or "",
            shutil.which("google-chrome") or "",
            shutil.which("chrome") or "",
            shutil.which("chromium") or "",
        ]
        for c in candidates:
            if c and os.path.exists(c):
                try:
                    subprocess.Popen([c, f"--app={base_url}"])
                    print(f"[Launcher] Tactical Mission Control opened in standalone app window ({os.path.basename(c)})")
                    print(f"  URL: {base_url}")
                    print("=" * 75)
                    return
                except Exception:
                    pass

    print(f"[Launcher] Opening Tactical Mission Control in default browser: {base_url}")
    webbrowser.open(base_url)
    print("=" * 75)


def trigger_mcu_reconnect(base_url: str) -> bool:
    """Tells the Pi debug server to re-detect and reconnect to all ESP32 USB serial ports."""
    url = f"{base_url}/api/mcu/command"
    payload = json.dumps({"action": "reconnect"}).encode("utf-8")
    print(f"[Launcher] Triggering remote ESP32 serial re-probe & reconnect on {base_url}...")
    try:
        req = urllib.request.Request(
            url,
            data=payload,
            headers={"Content-Type": "application/json", "User-Agent": "FolkraceLauncher/1.0"},
            method="POST",
        )
        with urllib.request.urlopen(req, timeout=4.0) as resp:
            data = json.loads(resp.read().decode("utf-8"))
            print(f"[Launcher] ESP32 Reconnect Status: {data.get('status', 'OK')} ({data.get('msg', 'reconnected')})")
            return True
    except Exception as e:
        print(f"[Launcher ERROR] Failed triggering ESP32 reconnect: {e}")
        return False


def download_database(base_url: str, output_path: Optional[str] = None) -> Optional[str]:
    """Downloads the active flight recorder SQLite database and returns its local path."""
    status = query_api_json(base_url, "/api/status")
    session_id = status.get("session_id", "recording") if status else "recording"
    
    if not output_path:
        out_dir = Path(ROOT_DIR) / "data" / "recordings"
        out_dir.mkdir(parents=True, exist_ok=True)
        output_path = str(out_dir / f"{session_id}.db")

    url = f"{base_url}/api/download"
    print(f"[Download] Fetching flight database from {url}...")
    t0 = time.time()
    try:
        req = urllib.request.Request(url, headers={"User-Agent": "FolkraceLauncher/1.0"})
        with urllib.request.urlopen(req, timeout=30.0) as resp, open(output_path, "wb") as f:
            total_bytes = 0
            while True:
                chunk = resp.read(65536)
                if not chunk:
                    break
                f.write(chunk)
                total_bytes += len(chunk)
        dt = time.time() - t0
        mb = total_bytes / (1024 * 1024)
        print(f"[Download] Saved {mb:.2f} MB in {dt:.2f}s ({mb/max(0.01, dt):.2f} MB/s) -> {output_path}")
        return output_path
    except Exception as e:
        print(f"[Download] Error: {e}")
        return None


def find_latest_local_recording() -> Optional[str]:
    """Returns the most recently modified .db file in data/recordings/, if any."""
    rec_dir = Path(ROOT_DIR) / "data" / "recordings"
    if not rec_dir.exists():
        return None
    dbs = sorted(rec_dir.glob("*.db"), key=lambda p: p.stat().st_mtime, reverse=True)
    return str(dbs[0]) if dbs else None


def run_local_replay_server(db_path: str, prefer_app_window: bool = True, open_browser: bool = True):
    """Starts a localhost 60 Hz offline replay server for a downloaded SQLite flight database."""
    from http.server import ThreadingHTTPServer, BaseHTTPRequestHandler
    from urllib.parse import urlparse
    import threading

    if PI4_DIR not in sys.path:
        sys.path.insert(0, PI4_DIR)
    tools_dir = os.path.join(ROOT_DIR, "tools")
    if tools_dir not in sys.path:
        sys.path.insert(0, tools_dir)
    from debug_recorder import TelemetryLogReader, add_label
    from debug_server import build_live_binary_packet
    from flight_analyzer import FlightSession

    html_path = os.path.join(PI4_DIR, "web_debug_gui.html")
    state_lock = threading.Lock()
    replay_state: Dict[str, Any] = {
        "db_path": db_path,
        "reader": TelemetryLogReader(db_path),
        "summary": None,
        "bounds": (1, 0, 0),
        "flight_session": None,
    }

    def _reload_db(new_path: str):
        new_reader = TelemetryLogReader(new_path)
        summary = new_reader.get_summary()
        bounds = new_reader.get_frame_bounds()
        with state_lock:
            old_reader = replay_state.get("reader")
            replay_state["db_path"] = new_path
            replay_state["reader"] = new_reader
            replay_state["summary"] = summary
            replay_state["bounds"] = bounds
            replay_state["flight_session"] = None
        if old_reader and hasattr(old_reader, "close"):
            old_reader.close()

    def _get_flight_session() -> FlightSession:
        with state_lock:
            sess = replay_state.get("flight_session")
            cur_db = replay_state["db_path"]
        if sess is None or sess.db_path != cur_db:
            new_sess = FlightSession(cur_db)
            with state_lock:
                if replay_state["db_path"] == cur_db:
                    replay_state["flight_session"] = new_sess
                sess = replay_state.get("flight_session") or new_sess
        return sess

    _reload_db(db_path)

    class LocalReplayHandler(BaseHTTPRequestHandler):
        protocol_version = "HTTP/1.1"

        def log_message(self, format, *args):
            pass

        def _send_bytes(self, body: bytes, content_type: str, status: int = 200):
            self.send_response(status)
            self.send_header("Content-Type", content_type)
            self.send_header("Content-Length", str(len(body)))
            self.send_header("Access-Control-Allow-Origin", "*")
            self.end_headers()
            self.wfile.write(body)

        def _send_json(self, obj: Any, status: int = 200):
            self._send_bytes(json.dumps(obj).encode("utf-8"), "application/json", status)

        def do_GET(self):
            path = urlparse(self.path).path
            if path in ("/", "/index.html"):
                with open(html_path, "rb") as f:
                    self._send_bytes(f.read(), "text/html")
            elif path == "/api/status":
                with state_lock:
                    cur_db = replay_state["db_path"]
                    summary = replay_state["summary"] or {}
                    min_id, max_id, _ = replay_state["bounds"]
                try:
                    _, _, disk_free = shutil.disk_usage(os.path.dirname(cur_db) or ROOT_DIR)
                    disk_free_gb = round(disk_free / (1024**3), 2)
                except Exception:
                    disk_free_gb = 0.0
                self._send_json({
                    "session_id": summary.get("session_id", os.path.basename(cur_db)),
                    "min_frame_id": min_id,
                    "total_recorded": max_id,
                    "dropped_frames": 0,
                    "pruned_frames": 0,
                    "driving_seconds": summary.get("driving_seconds", 0.0),
                    "driving_minutes": summary.get("driving_minutes", 0.0),
                    "max_buffer_minutes": 30.0,
                    "file_size_mb": summary.get("file_size_mb", 0.0),
                    "disk_free_gb": disk_free_gb,
                    "temp_c": 0.0,
                    "loop_hz": 60.0,
                    "cpu_affinity": [],
                })
            elif path.startswith("/api/frame_bin/"):
                try:
                    frame_id = int(path.split("/")[-1])
                except ValueError:
                    self.send_error(400, "Invalid frame ID")
                    return
                with state_lock:
                    reader = replay_state["reader"]
                frame_data = reader.get_frame(frame_id)
                if not frame_data:
                    self.send_error(404, f"Frame {frame_id} not found")
                    return
                try:
                    self._send_bytes(build_live_binary_packet(frame_data), "application/octet-stream")
                except Exception as e:
                    self.send_error(500, f"Failed to pack frame {frame_id}: {e}")
            elif path.startswith("/api/frame/"):
                try:
                    frame_id = int(path.split("/")[-1])
                except ValueError:
                    self._send_json({"error": "Invalid frame ID"}, 400)
                    return
                with state_lock:
                    reader = replay_state["reader"]
                frame_data = reader.get_frame(frame_id)
                if frame_data:
                    self._send_json(frame_data)
                else:
                    self._send_json({"error": f"Frame {frame_id} not found"}, 404)
            elif path == "/api/anomalies":
                with state_lock:
                    reader = replay_state["reader"]
                self._send_json(reader.get_anomalies())
            elif path == "/api/ai_report":
                try:
                    self._send_json(_get_flight_session().build_digest())
                except Exception as e:
                    self._send_json({"error": str(e)}, 500)
            elif path == "/api/episodes":
                try:
                    self._send_json(_get_flight_session().extract_episodes())
                except Exception as e:
                    self._send_json({"error": str(e)}, 500)
            elif path == "/api/labels":
                with state_lock:
                    reader = replay_state["reader"]
                self._send_json(reader.get_labels())
            elif path == "/api/download":
                with state_lock:
                    cur_db = replay_state["db_path"]
                if os.path.exists(cur_db):
                    with open(cur_db, "rb") as f:
                        data = f.read()
                    self.send_response(200)
                    self.send_header("Content-Type", "application/octet-stream")
                    self.send_header("Content-Disposition", f'attachment; filename="{os.path.basename(cur_db)}"')
                    self.send_header("Content-Length", str(len(data)))
                    self.end_headers()
                    self.wfile.write(data)
                else:
                    self.send_error(404, "Database not found")
            elif path in ("/api/live", "/api/live_bin"):
                self._send_json({"status": "REPLAY_OFFLINE"}, 200) if path == "/api/live" else self.send_error(404)
            else:
                self._send_json({"error": "Not Found"}, 404)

        def do_POST(self):
            path = urlparse(self.path).path
            if path == "/api/upload_db":
                content_len = int(self.headers.get("Content-Length", 0))
                if content_len <= 0:
                    self._send_json({"error": "Empty body"}, 400)
                    return
                body = self.rfile.read(content_len)
                raw_name = os.path.basename(self.headers.get("X-Filename", "uploaded.db")) or "uploaded.db"
                out_dir = Path(ROOT_DIR) / "data" / "recordings"
                out_dir.mkdir(parents=True, exist_ok=True)
                target_path = str(out_dir / raw_name)
                with open(target_path, "wb") as f:
                    f.write(body)
                try:
                    _reload_db(target_path)
                    self._send_json({"status": "ok", "db_path": target_path})
                except Exception as e:
                    self._send_json({"error": str(e)}, 400)
            elif path == "/api/labels":
                content_len = int(self.headers.get("Content-Length", 0))
                if content_len <= 0:
                    self._send_json({"error": "Empty body"}, 400)
                    return
                try:
                    payload = json.loads(self.rfile.read(content_len).decode("utf-8"))
                    start_f = int(payload["start_frame"])
                    end_f = int(payload["end_frame"])
                    tag = str(payload["tag"]).strip()
                    if not tag:
                        raise ValueError("tag must be non-empty")
                    note = str(payload.get("note", ""))
                    author = str(payload.get("author", "user"))
                    with state_lock:
                        cur_db = replay_state["db_path"]
                    lbl_id = add_label(cur_db, start_f, end_f, tag, note=note, author=author)
                    with state_lock:
                        replay_state["flight_session"] = None
                    self._send_json({"status": "ok", "label_id": lbl_id})
                except Exception as e:
                    self._send_json({"error": str(e)}, 400)
            else:
                self._send_json({"error": "Endpoint not found"}, 404)

    httpd = ThreadingHTTPServer(("127.0.0.1", 0), LocalReplayHandler)
    port = httpd.server_address[1]
    url = f"http://127.0.0.1:{port}/?replay=1"
    print(f"[Replay] Serving {db_path} locally at 60 Hz: {url}")
    print("[Replay] Press Ctrl+C to stop the local replay server.")
    if open_browser:
        launch_tactical_gui(url, prefer_app_window=prefer_app_window)
    try:
        httpd.serve_forever()
    except KeyboardInterrupt:
        print("\n[Replay] Stopped.")
    finally:
        httpd.server_close()


def run_full_verification(base_url: str, ssh_host: str = "robot") -> bool:
    """
    Runs comprehensive end-to-end verification of all Pi server endpoints.
    """
    print("\n" + "=" * 75)
    print(f"  FOLKRACE PI 4: FULL SUBSYSTEM VERIFICATION REPORT")
    print("=" * 75)
    print(f"  Target Server Base URL: {base_url}")
    print(f"  SSH Host Target       : {ssh_host}")
    print("-" * 75)

    all_passed = True

    return mod


def list_local_recordings(directories=SEARCH_DIRS) -> List[Path]:
    """Returns all .db files in the given directories sorted newest first."""
    dbs: List[Path] = []
    seen = set()
    for directory in directories:
        if not directory.exists():
            continue
        for p in directory.glob("*.db"):
            resolved = p.resolve()
            if resolved not in seen:
                seen.add(resolved)
                dbs.append(p)
    return sorted(dbs, key=lambda p: p.stat().st_mtime, reverse=True)


def get_latest_downloads_recording() -> Optional[Path]:
    """Returns the newest .db file in Downloads, or None if none exist."""

This directory contains consolidated, human-facing one-click launchers for 3D physics simulation, Genetic Algorithm parameter optimization, Raspberry Pi 4 flight operations, and physical microcontroller bench testing (ESP32 and STM32).

> [!NOTE]

## 1. Ordering & Future Renumbering

Launchers use leading numeric prefixes (`1_`–`9_`) to preserve workflow order in file explorers and terminals:
* **`1–3` — Simulation & Training**: Webots 3D previews and GA / CMA-ES optimization dashboard.
* **`4–5` — Pi 4 Operations & Replay**: Live Tactical Mission Control and offline 60 Hz flight log replay.
* **`6–8` — ESP32 Benches**: Quad VL53L8CX ToF Hub, BNO086 9-DOF IMU dashboard, and 333 Hz digital steering servo.
* **`9` — STM32 Bench**: Dual-motor direct-drive FOC & MT6816 magnetic encoders.

## 3. Quick Usage Examples

### Running 3D Simulation with Curvilinear SL-MPC:
```bash
python 1_launchers/1_champion_preview.py
```
* Launches Webots with the dynamic obstacle world (`track_racing.wbt`) and companion 2D telemetry radar GUI.
* Use `python 1_launchers/2_fast_champion_drive.py` for maximum real-time FPS without GUI/UDP overhead.

### Raspberry Pi 4 Dual-Core Debug Studio & Offline Replay:
```bash
python 1_launchers/4_pi4_debug_studio.py
python 1_launchers/5_local_flight_replay.py
```
* Auto-detects the Pi 4 (`robot`), opens Web Tactical Mission Control (`http://<pi>:8080`), and supports flight log downloads and offline 60 Hz scrubbing.

### Benchmarking Quad VL53L8CX ToF Array:
```bash
python 1_launchers/6_esp32_tof_gui.py
```
* Connects to the ESP32 ToF Hub over USB serial (921,600 baud) and visualizes all 4 ToF sensors (16 ray projections, multi-target status, and SPAD counts).

### High-Rate Digital Steering Servo Calibration:
```bash
python 1_launchers/8_esp32_servo_gui.py
```
* Connects to ESP32 Main over USB serial to tune center trim, deflection limits, and 333 Hz response dynamics.

### Benchmarking Physical STM32 Direct-Drive Motors:
```bash
python 1_launchers/9_stm32_encoder_motor_gui.py
```
* Connects over native USB CDC or UART, visualizes MT6816 magnetic angle and RPM response, and runs automated 128-point LUT calibration.
