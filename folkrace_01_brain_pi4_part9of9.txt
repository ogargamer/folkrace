def parse_pinout_database(csv_text):
    database = {}
    lines = csv_text.strip().split('\n')
    for line in lines:
        line = line.strip()
        if not line or line.startswith("Header_or_Connector"):
            continue
        reader = csv.reader([line])
        try:
            row = next(reader)
        except Exception:
            continue
        if len(row) < 12:
            continue
            
        header_conn = row[0].strip()
        phys_pin = row[1].strip()
        signal_name = row[2].strip()
        gpio_num_str = row[3].strip()
        pin_type = row[4].strip()
        primary_func = row[5].strip()
        alt0 = row[6].strip()
        alt1 = row[7].strip()
        alt2 = row[8].strip()
        alt3 = row[9].strip()
        alt4 = row[10].strip()
        notes = row[11].strip()

        try:
            gpio_num = int(gpio_num_str.replace("GPIO", ""))
        except ValueError:
            gpio_num = None

        pin_info = {
            "Header": header_conn,
            "PhysicalPin": phys_pin,
            "Signal": signal_name,
            "GPIO": gpio_num,
            "Type": pin_type,
            "PrimaryFunc": primary_func,
            "ALT0": alt0,
            "ALT1": alt1,
            "ALT2": alt2,
            "ALT3": alt3,
            "ALT4": alt4,
            "Notes": notes
        }
        
        if header_conn == "40-Pin Header":
            database[f"PIN_{phys_pin}"] = pin_info
            database[phys_pin] = pin_info
        
        if gpio_num is not None:
            database[f"GPIO{gpio_num}"] = pin_info
            database[str(gpio_num)] = pin_info
            
        database[signal_name] = pin_info
        
    return database

PIN_DB = parse_pinout_database(RAW_PINOUT_CSV)

@dataclass(frozen=True)
class PinRequest:
    """One exclusive physical-wire assignment in the Pi pinout CSP."""

    name: str
    candidates: Tuple[int, ...]
    function: str
    description: str


class PinConstraintError(RuntimeError):
    """Raised when the requested hardware cannot be assigned uniquely."""


class PinoutSolver:
    """Small exact CSP solver for exclusive GPIO assignments.

    Each request is a node in a bipartite graph (signal -> allowed GPIOs).  The
    backtracking search uses the minimum-remaining-values heuristic and retains
    the lowest-cost complete matching, where earlier candidates are preferred.
    Shared-bus wires are represented once, so they are intentionally shared by
    devices without making two logical assignments to one physical pin.
    """

    def __init__(self, valid_gpios: Iterable[int], reserved_gpios: Iterable[int]):
        self.valid_gpios = set(valid_gpios)
        self.reserved_gpios = set(reserved_gpios)
        self._owners: Dict[int, str] = {}

    def reserve(self, request: PinRequest) -> None:
        if len(request.candidates) != 1:
            raise ValueError(f"Fixed request {request.name} needs exactly one GPIO")
        gpio = request.candidates[0]
        if gpio not in self.valid_gpios:
            raise PinConstraintError(f"{request.name}: GPIO{gpio} is not on the 40-pin header")
        if gpio in self.reserved_gpios:
            raise PinConstraintError(f"{request.name}: GPIO{gpio} is reserved by the system")
        if gpio in self._owners:
            raise PinConstraintError(
                f"GPIO{gpio} collision: {request.name} conflicts with {self._owners[gpio]}"
            )
        self._owners[gpio] = request.name

    def solve(self, requests: Sequence[PinRequest]) -> Dict[str, int]:
        candidates: Dict[str, List[int]] = {}
        request_by_name = {request.name: request for request in requests}
        if len(request_by_name) != len(requests):
            raise PinConstraintError("Every pin-request name must be unique")

        for request in requests:
            options = [
                gpio for gpio in request.candidates
                if gpio in self.valid_gpios
                and gpio not in self.reserved_gpios
                and gpio not in self._owners
            ]
            if not options:
                raise PinConstraintError(
                    f"No available GPIO candidates remain for {request.name}: {request.candidates}"
                )
            candidates[request.name] = options

        best_assignment: Dict[str, int] | None = None
        best_cost: int | None = None

        def search(pending: Tuple[str, ...], chosen: Dict[str, int], cost: int) -> None:
            nonlocal best_assignment, best_cost
            if best_cost is not None and cost >= best_cost:
                return
            if not pending:
                best_assignment = dict(chosen)
                best_cost = cost
                return

            # Minimum remaining values: expose impossible / over-subscribed
            # configurations early instead of printing a misleading report.
            name = min(
                pending,
                key=lambda item: sum(
                    gpio not in self._owners and gpio not in chosen.values()
                    for gpio in candidates[item]
                ),
            )
            request = request_by_name[name]
            remaining = tuple(item for item in pending if item != name)

            for preference_cost, gpio in enumerate(candidates[name]):
                if gpio in self._owners or gpio in chosen.values():
                    continue
                chosen[name] = gpio
                search(remaining, chosen, cost + preference_cost)
                del chosen[name]

        search(tuple(request_by_name), {}, 0)
        if best_assignment is None:
            detail = ", ".join(f"{name}={options}" for name, options in candidates.items())
            raise PinConstraintError(f"No one-to-one GPIO assignment exists ({detail})")

        for name, gpio in best_assignment.items():
            self._owners[gpio] = name
        return best_assignment

    @property
    def owners(self) -> Dict[int, str]:
        return dict(self._owners)


HEADER_GPIOS = tuple(range(0, 28))


def request(name: str, gpio: int, function: str, description: str) -> PinRequest:
    """Declare a fixed assignment and route it through the collision checker."""
    return PinRequest(name, (gpio,), function, description)


def physical_pin(gpio: int) -> int:
    info = PIN_DB.get(f"GPIO{gpio}")
    if not info or info["PhysicalPin"].isdigit() is False:
        raise PinConstraintError(f"GPIO{gpio} has no 40-pin-header entry in the pin database")
    return int(info["PhysicalPin"])


def solve_pi_pinout(use_usb_cables: bool = True, enable_hardware_tof_uart: bool = False):
    """Return a collision-free assignment for this Pi 4/5 hardware topology.

    Hardware Connections:
      Dual USB Cable Mode (Recommended / Default):
        - ESP32-Main: Direct USB Cable to Pi 4 USB Port (/dev/ttyUSB0 or /dev/ttyACM0 @ 921,600 baud)
        - ESP32-ToF:  Direct USB Cable to Pi 4 USB Port (/dev/ttyUSB1 or /dev/ttyACM1 @ 921,600 baud)
        - Zero GPIO pins or flying wires consumed on the 40-pin header!

      Legacy GPIO UART Mode:
        1. Primary High-Speed UART (/dev/ttyAMA0): Duplex link to ESP32-Main MCU (Chassis Telemetry @ 921,600 baud)
        2. Handshake Interrupt: GPIO20 Input triggered by ESP32-Main batch-ready pulse
        3. Secondary UART2 (/dev/ttyAMA2): Hardware UART to ESP32-ToF Hub (@ 921,600 baud)
    """

    reserved = set()
    solver = PinoutSolver(HEADER_GPIOS, reserved)

    if use_usb_cables:
        # Dual USB cable mode: 0 header pins assigned to MCU comms
        return {}

    fixed_requests = [
        # ESP32-Main Telemetry Highway (Primary UART0)
        request("ESP32_MAIN_UART_TX", 14, "UART0_TX", "Duplex TX -> ESP32-Main RX (GPIO 44) @ 921,600 baud"),
        request("ESP32_MAIN_UART_RX", 15, "UART0_RX", "Duplex RX <- ESP32-Main TX (GPIO 43) @ 921,600 baud"),
        request("ESP32_MAIN_DRDY",    20, "GPIO20",   "Batch data-ready handshake pulse from ESP32-Main (GPIO 1)"),
    ]

    if enable_hardware_tof_uart:
        fixed_requests.extend([
            # ESP32-ToF Hub Telemetry Link (Secondary UART2 via dtoverlay=uart2)
            request("ESP32_TOF_UART_TX", 0, "UART2_TX", "Secondary TX -> ESP32-ToF RX (GPIO 44) @ 921,600 baud"),
            request("ESP32_TOF_UART_RX", 1, "UART2_RX", "Secondary RX <- ESP32-ToF TX (GPIO 43) @ 921,600 baud"),
        ])

    for fixed in fixed_requests:
        solver.reserve(fixed)

    requests_by_name = {item.name: item for item in fixed_requests}
    assignments = {
        item.name: item.candidates[0]
        for item in fixed_requests
    }

    # This is deliberately an assertion, not just a summary: regressions that
    # reintroduce a duplicate GPIO now fail in development and CI.
    if len(assignments) != len(set(assignments.values())):
        raise PinConstraintError("Internal error: duplicate GPIO survived the solver")

    return {
        name: (
            gpio,
            physical_pin(gpio),
            requests_by_name[name].function,
            requests_by_name[name].description,
        )
        for name, gpio in assignments.items()
    }


def print_report(use_usb_cables: bool = True) -> None:
    assignments = solve_pi_pinout(use_usb_cables=use_usb_cables, enable_hardware_tof_uart=not use_usb_cables)
    used = {gpio for gpio, _, _, _ in assignments.values()}
    free = sorted(set(HEADER_GPIOS) - used)

    print("\n" + "=" * 100)
    print("  RASPBERRY PI 4 / 5 PINOUT CSP - VERIFIED COLLISION-FREE ASSIGNMENT")
    mode_str = "DUAL USB CABLE ARCHITECTURE (ZERO HEADER PINS USED)" if use_usb_cables else "LEGACY GPIO UART WIRING"
    print(f"  Topology Mode: {mode_str}")
    print("=" * 100)
    if use_usb_cables:
        print("  [USB SERIAL ARCHITECTURE]")
        print("  * ESP32-Main : Connected via USB Cable -> /dev/ttyUSB0 or /dev/ttyACM0 (921,600 baud)")
        print("  * ESP32-ToF  : Connected via USB Cable -> /dev/ttyUSB1 or /dev/ttyACM1 (921,600 baud)")
        print("  * Header Pins: 0 jumper wires on 40-pin header. 100% plug-and-play.")
        print("  " + "-" * 96)
        print(f"  Assigned Header GPIOs: 0 pins")
        print(f"  Free Available Header GPIOs ({len(free)}/28 pins): {free}")
    else:
        print(f"  {'Signal':<21} | {'Physical pin':<14} | {'GPIO':<9} | {'Function':<12} | Description")
        print("  " + "-" * 96)
        for name, (gpio, pin, function, description) in sorted(assignments.items(), key=lambda item: item[1][1]):
            print(f"  {name:<21} | Pin {pin:<10} | GPIO {gpio:<4} | {function:<12} | {description}")
        print("  " + "-" * 96)
        print(f"  Assigned GPIOs: {len(used)} unique pins - verified collision-free")
        print(f"  Free Available GPIOs ({len(free)} pins): {free}")
    print("=" * 100 + "\n")


def main() -> None:
    import argparse
    parser = argparse.ArgumentParser(description="Raspberry Pi 4 Pinout Solver")
    parser.add_argument("--gpio-uart", action="store_true", help="Show legacy GPIO header UART wiring instead of USB cables")
    args = parser.parse_args()
    print_report(use_usb_cables=not args.gpio_uart)


if __name__ == "__main__":
    main()

"""
================================================================================
  FOLKRACE PI 4: DUAL-CORE AUTONOMOUS SYSTEM LAUNCHER
================================================================================
  Spawns and monitors:
    1. Path Planner Pipeline -> HARD PINNED TO CPU CORE 1
    2. Debug Server & Black Box Recorder -> HARD PINNED TO CPU CORE 2 (and 3)

import os
import sys
import time
import signal
import subprocess
import platform

CURR_DIR = os.path.dirname(os.path.abspath(__file__))
if CURR_DIR not in sys.path:
    sys.path.insert(0, CURR_DIR)

PLANNER_SCRIPT = os.path.join(CURR_DIR, "path_planner_runner.py")
DEBUG_SERVER_SCRIPT = os.path.join(CURR_DIR, "debug_server.py")
BRIDGE_SCRIPT = os.path.join(CURR_DIR, "mcu_uart_bridge.py")


def check_core_assignment(pid: int) -> str:
    """Reads /proc/<pid>/status to verify assigned CPU cores."""
    status_file = f"/proc/{pid}/status"
    if os.path.exists(status_file):
        try:
            with open(status_file, "r") as f:
                for line in f:
                    if line.startswith("Cpus_allowed_list:"):
                        return line.split(":", 1)[1].strip()
        except Exception:
            pass
    return "unknown"


def main():
    def _handle_sigterm(_signum, _frame):
        raise KeyboardInterrupt()

    signal.signal(signal.SIGTERM, _handle_sigterm)

    print("=" * 70)
    print("  FOLKRACE PI 4: DUAL-CORE HARDWARE ISOLATION LAUNCHER")
    print("=" * 70)
    print(f"  Platform Architecture : {platform.machine()} ({sys.platform})")
    print(f"  Available CPU Cores   : {os.cpu_count()}")
    print("=" * 70)

    has_taskset = False
    if sys.platform == "linux":
        try:
            has_taskset = bool(subprocess.run(["which", "taskset"], stdout=subprocess.PIPE, text=True).stdout.strip())
        except Exception:
            pass

    # Launch MCU UART Bridge on Core 3 (if available)
    p_bridge = None
    if os.path.exists(BRIDGE_SCRIPT):
        print("[1/3] Launching MCU Binary UART Bridge on Core 3...")
        if has_taskset:
            cmd_bridge = ["taskset", "-c", "3", sys.executable, BRIDGE_SCRIPT]
        else:
            cmd_bridge = [sys.executable, BRIDGE_SCRIPT]
        p_bridge = subprocess.Popen(cmd_bridge, cwd=CURR_DIR)
        time.sleep(0.5)
        print(f"      PID: {p_bridge.pid} | CPU Core Mask: {check_core_assignment(p_bridge.pid)}")

    # Launch Debug Server on Core 2
    print("[2/3] Launching Debug Server on Core 2...")
    if has_taskset:
        cmd_debug = ["taskset", "-c", "2", sys.executable, DEBUG_SERVER_SCRIPT]
    else:
        cmd_debug = [sys.executable, DEBUG_SERVER_SCRIPT]

    p_debug = subprocess.Popen(cmd_debug, cwd=CURR_DIR)
    time.sleep(1.0)
    print(f"      PID: {p_debug.pid} | CPU Core Mask: {check_core_assignment(p_debug.pid)}")

    # Launch Path Planner on Core 1
    print("[3/3] Launching Real-Time Path Planner on Core 1...")
    if sys.platform == "linux":
        cmd_planner = ["taskset", "-c", "1", sys.executable, PLANNER_SCRIPT]
    else:
        cmd_planner = [sys.executable, PLANNER_SCRIPT]

    p_planner = subprocess.Popen(cmd_planner, cwd=CURR_DIR)
    time.sleep(1.0)
    print(f"      PID: {p_planner.pid} | CPU Core Mask: {check_core_assignment(p_planner.pid)}")

    print("=" * 70)
    print("  [SYSTEM READY]")

"""
================================================================================
  FOLKRACE PI 4: TOF EGO-MOTION LATENCY COMPENSATOR (OPTION 1)
================================================================================
  Compensates for the ~16.6ms ToF sensor integration & UART communication delay
  using high-speed chassis telemetry (fused forward velocity v_est and yaw rate
  gyro_z from ESP32-Main).

  Kinematics:
    For each beam i with mount offset (x0_i, y0_i) and heading theta_i (where +Y
    is forward and +X is right):
      delta_y   = v_est * dt
      delta_psi = gyro_z * dt
      delta_r_i = [v_est * sin(theta_i) + gyro_z * (x0_i * sin(theta_i) - y0_i * cos(theta_i))] * dt
      r_compensated_i = r_measured_i - delta_r_i

    - Precomputes all trigonometric coefficients at instantiation (zero allocation).
    - Sub-microsecond execution time (< 0.001 ms for all 16 beams).
    - Preserves open-air free space (beams at max_range remain clear).
================================================================================
"""

import math
from typing import Dict, List, Tuple, Union, Optional

# 16 Front ToF Sensors matching smart_hybrid_costmap_front.cpp & vehicle_config.json
# Format: (name, x0, y0, heading_rad, min_range, max_range)
SENSOR_RIG_CONFIG = [
    # Front-Left Array (cross-pointing Right)
    ("tof_front_left_4.1deg",  -0.058, 0.092, 1.5512, 0.001, 2.40),
    ("tof_front_left_15.4deg", -0.058, 0.092, 1.3548, 0.001, 2.40),
    ("tof_front_left_26.6deg", -0.058, 0.092, 1.1585, 0.001, 2.40),
    ("tof_front_left_37.9deg", -0.058, 0.092, 0.9621, 0.001, 2.40),
    ("tof_front_left_49.1deg", -0.057, 0.090, 0.7658, 0.001, 2.40),
    ("tof_front_left_60.4deg", -0.057, 0.090, 0.5694, 0.001, 2.40),
    ("tof_front_left_71.6deg", -0.057, 0.090, 0.3731, 0.001, 2.40),
    ("tof_front_left_82.9deg", -0.057, 0.090, 0.1767, 0.001, 2.40),
    # Front-Right Array (cross-pointing Left)
    ("tof_front_right_4.1deg",   0.058, 0.092, 1.5904, 0.001, 2.40),
    ("tof_front_right_15.4deg",  0.058, 0.092, 1.7868, 0.001, 2.40),
    ("tof_front_right_26.6deg",  0.058, 0.092, 1.9831, 0.001, 2.40),
    ("tof_front_right_37.9deg",  0.058, 0.092, 2.1795, 0.001, 2.40),
    ("tof_front_right_49.1deg",  0.057, 0.090, 2.3758, 0.001, 2.40),
    ("tof_front_right_60.4deg",  0.057, 0.090, 2.5722, 0.001, 2.40),
    ("tof_front_right_71.6deg",  0.057, 0.090, 2.7685, 0.001, 2.40),
    ("tof_front_right_82.9deg",  0.057, 0.090, 2.9649, 0.001, 2.40),
]

SENSOR_NAMES = [cfg[0] for cfg in SENSOR_RIG_CONFIG]
DEFAULT_NOMINAL_LATENCY_S = 0.01667  # 16.67 ms (60 Hz frame interval)


class ToFLatencyCompensator:
    """Instantaneous Ego-Motion Latency Compensator for 16-channel ToF sensor rig."""

    def __init__(self, default_dt_s: float = DEFAULT_NOMINAL_LATENCY_S):
        self.default_dt_s = float(default_dt_s)

        # Precompute per-beam geometric constants
        self.sensor_names: List[str] = []
        self.coeff_fwd: List[float] = []   # sin(heading)
        self.coeff_rot: List[float] = []   # x0 * sin(heading) - y0 * cos(heading)
        self.min_ranges: List[float] = []
        self.max_ranges: List[float] = []

        for name, x0, y0, heading, min_r, max_r in SENSOR_RIG_CONFIG:
            sin_h = math.sin(heading)
            cos_h = math.cos(heading)
            c_fwd = sin_h
            c_rot = x0 * sin_h - y0 * cos_h

            self.sensor_names.append(name)
            self.coeff_fwd.append(c_fwd)
            self.coeff_rot.append(c_rot)
            self.min_ranges.append(min_r)
            self.max_ranges.append(max_r)

    def compensate_list(
        self,
        ranges: List[float],
        v_est: float,
        gyro_z: float,
        dt: Optional[float] = None,
    ) -> List[float]:
        """
        Compensates a list of 16 ranges (in meters) for robot motion during latency dt.
        Returns a new list of 16 motion-compensated ranges.
        """
        step_dt = self.default_dt_s if (dt is None or dt <= 0.0) else float(dt)
        n = min(len(ranges), 16)
        out = list(ranges)

        for i in range(n):
            r = ranges[i]
            max_r = self.max_ranges[i]
            # If beam is clear / open-air, do not contract into phantom obstacle
            if r >= max_r - 0.015:
                continue

            delta_r = (v_est * self.coeff_fwd[i] + gyro_z * self.coeff_rot[i]) * step_dt
            r_comp = r - delta_r
            min_r = self.min_ranges[i]
            if r_comp < min_r:
                r_comp = min_r
            elif r_comp > max_r:
                r_comp = max_r
            out[i] = round(r_comp, 4)

        return out

    def compensate_dict(
        self,
        readings: Dict[str, float],
        v_est: float,
        gyro_z: float,
        dt: Optional[float] = None,
    ) -> Dict[str, float]:
        """
        Compensates a dict of {sensor_name: range_m} for robot motion during latency dt.
        Returns a new dictionary with motion-compensated ranges.
        """
        step_dt = self.default_dt_s if (dt is None or dt <= 0.0) else float(dt)
        out = dict(readings)

        for i, name in enumerate(self.sensor_names):
            if name not in readings:
                continue
            r = float(readings[name])
            max_r = self.max_ranges[i]
            if r >= max_r - 0.015:
                continue

            delta_r = (v_est * self.coeff_fwd[i] + gyro_z * self.coeff_rot[i]) * step_dt
            r_comp = r - delta_r
            min_r = self.min_ranges[i]
            if r_comp < min_r:
                r_comp = min_r
            elif r_comp > max_r:
                r_comp = max_r
            out[name] = round(r_comp, 4)

        return out

    def compute_projected_anchor(
        self,
        v_est: float,
        gyro_z: float,
        dt: Optional[float] = None,
    ) -> Tuple[float, float, float]:
        """
        Computes the projected robot pose (anchor_x, anchor_y, anchor_yaw) in the
        sensor coordinate frame elapsed over latency dt:
          - anchor_y: forward displacement (m)
          - anchor_x: lateral displacement (m)
          - anchor_yaw: heading change (rad)
        Used directly by costmap_scout_gaps_projected and SL-MPC initial state (x0, y0, psi0).
        """
        step_dt = self.default_dt_s if (dt is None or dt <= 0.0) else float(dt)
        dy = v_est * step_dt
        dtheta = gyro_z * step_dt
        # Lateral displacement in circular turning arc: 0.5 * v * omega * dt^2
        dx = 0.5 * v_est * gyro_z * (step_dt ** 2)
        return (dx, dy, dtheta)

#!/usr/bin/env python3
"""
Automatically puts STM32 into System ROM DFU mode without physical buttons.
Identifies STMicroelectronics USB CDC port (VID 0x0483, PID 0x5740),
sends 'DFU\\n' to trigger firmware software jump to bootloader, and optionally
executes PlatformIO upload.
"""

import os
import sys
import time
import argparse
import subprocess
import serial
import serial.tools.list_ports

def find_stm32_port():
    for p in serial.tools.list_ports.comports():
        if p.vid == 0x0483 and p.pid == 0x5740:
            return p.device
        if "STM" in (p.description or "") or "STMicroelectronics" in (p.manufacturer or ""):
            return p.device
    return None

def trigger_dfu(port=None):
    if not port:
        port = find_stm32_port()
    if not port:
        print("[ERR] No active STM32 USB CDC port found (VID: 0x0483, PID: 0x5740).")
        print("      Device may already be in DFU mode or disconnected.")
        return False

    print(f"[INFO] Found STM32 on {port}. Sending DFU command...")
    try:
        ser = serial.Serial(port, baudrate=921600, timeout=1.0)
        ser.dtr = True
        ser.rts = True
        time.sleep(0.05)
        ser.write(b"\n\nDFU\n")
        ser.flush()
        time.sleep(0.2)
        ser.close()
        print(f"[OK] DFU trigger sent. Device on {port} is rebooting to ST System DFU Bootloader...")
        time.sleep(3.0)
        return True
    except Exception as e:
        print(f"[WARN] Error sending DFU command to {port}: {e}")
        return False

def main():
    parser = argparse.ArgumentParser(description="Reboot STM32 into DFU mode and optionally upload firmware.")
    parser.add_argument("--port", "-p", help="Explicit COM port (e.g. COM11). Auto-detects if omitted.")
    parser.add_argument("--upload", "-u", action="store_true", help="Run 'pio run -e stm32_dual_foc -t upload' after rebooting to DFU.")
    parser.add_argument("--env", "-e", default="stm32_dual_foc", help="PlatformIO environment name (default: stm32_dual_foc)")
    args = parser.parse_args()

    success = trigger_dfu(args.port)

    if args.upload:
        if os.path.exists("platformio.ini"):
            cmd = ["pio", "run", "-e", args.env, "-t", "upload"]
        else:
            cmd = ["pio", "run", "-d", "4_stm32", "-e", args.env, "-t", "upload"]
        print(f"\n[RUN] Running: {' '.join(cmd)}")
        ret = subprocess.run(cmd)
        sys.exit(ret.returncode)

if __name__ == "__main__":
    main()

- **Steering (`cmd_steer_rad`, `steer_deg`) & Yaw (`yaw`, `gyro_z`)**:

  - Negative (`-`) = **Steer Right / CW**

  - Opening on **Right (`X > 0`)**: forward (`v > 0.1 m/s`) steer `-37.2°` (Right); reverse (`v <= 0.1 m/s`) steer `+37.2°` (Left).

## 4. Hardware Pinout Reference (STM32G474 LQFP48 Dual-Motor FOC)

Each STM32 board controls two direct-drive brushless motors and two MT6816 magnetic angle encoders using dedicated hardware timers and SPI peripherals:

| Pin | Function / Role | Peripheral | Notes |
| :--- | :--- | :--- | :--- |
| **PA1** | Motor Phase B (M1) | TIM2_CH2 | Center-aligned 25 kHz PWM |
| **PA2** | Motor Phase C (M1) | TIM2_CH3 | Center-aligned 25 kHz PWM |
| **PA3** | Motor 1 Driver Enable | GPIO Out | Active-High driver enable |
| **PA4** | Motor Phase B (M2) | TIM3_CH2 | Center-aligned 25 kHz PWM |
| **PA5** | Motor Phase A (M1) | TIM2_CH1 | Center-aligned 25 kHz PWM |
| **PA6** | Motor Phase A (M2) | TIM3_CH1 | Center-aligned 25 kHz PWM |
| **PA7** | Motor 2 Driver Enable | GPIO Out | Active-High driver enable |
| **PA9** | Telemetry TX | USART1_TX | Communication link |
| **PA10** | Telemetry RX | USART1_RX | Communication link |
| **PA11** | USB D- | Native USB | DFU Bootloader / CDC Serial |
| **PA12** | USB D+ | Native USB | DFU Bootloader / CDC Serial |
| **PA13** | SWDIO | SWD SYS | Hardware debug data |
| **PA14** | SWCLK | SWD SYS | Hardware debug clock |
| **PA15** | Motor 1 Encoder CS | GPIO Out | SPI1 Chip Select for MT6816 #1 |
| **PB0** | Motor Phase C (M2) | TIM3_CH3 | Center-aligned 25 kHz PWM |
| **PB3** | Encoder 1 SCK | SPI1_SCK | MT6816 #1 SPI Clock |
| **PB4** | Encoder 1 MISO | SPI1_MISO | MT6816 #1 SPI Data In |
| **PB5** | Encoder 1 MOSI | SPI1_MOSI | MT6816 #1 SPI Data Out |
| **PB11** | Motor 2 Encoder CS | GPIO Out | SPI2 Chip Select for MT6816 #2 |
| **PB13** | Encoder 2 SCK | SPI2_SCK | MT6816 #2 SPI Clock (AF5) |
| **PB14** | Encoder 2 MISO | SPI2_MISO | MT6816 #2 SPI Data In (AF5) |
| **PB15** | Encoder 2 MOSI | SPI2_MOSI | MT6816 #2 SPI Data Out (AF5) |
| **PC13** | Onboard Status LED | GPIO Out | Active-Low user status LED |
